## 二、聚宽策略骨架（感受事件驱动风格）

```python
# ==========================================
# 🌟 聚宽策略模板(在网页编辑器里运行)
# ==========================================
def initialize(context):
    # 🚀 初始化：只运行一次
    g.stock_pool = get_index_stocks("000300.XSHG")   # 沪深300成分
    set_option("use_real_price", True)               # ⚠️ 必开！用真实价格回测
    set_order_cost(OrderCost(open_tax=0, close_tax=0.0005,
                             open_commission=0.00025,
                             close_commission=0.00025,
                             min_commission=5), type="stock")  # 显式设成本
    run_monthly(rebalance, 1, time="10:00")          # 每月第一个交易日10点调仓

def rebalance(context):
    # 📌 调仓逻辑：12-1动量选前20%
    prices = get_price(g.stock_pool, count=252,
                       fields=["close"], panel=False)
    mom = prices.groupby("code").apply(
        lambda g: g["close"].iloc[-21] / g["close"].iloc[0] - 1)
    picks = mom.nlargest(int(len(mom) * 0.2)).index.tolist()

    # 先卖后买，腾出资金
    for code in list(context.portfolio.positions):
        if code not in picks:
            order_target(code, 0)
    weight = 1.0 / len(picks)
    for code in picks:
        order_target_value(code, context.portfolio.total_value * weight)
```
