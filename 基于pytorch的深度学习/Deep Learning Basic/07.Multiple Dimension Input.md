```
import numpy as np
import torch
from sklearn.datasets import load_diabetes

# 1. 加载 sklearn 糖尿病数据集（10个特征）
diabetes = load_diabetes()
x_numpy = diabetes.data
y_numpy = diabetes.target

x_data = torch.from_numpy(x_numpy).float()
y_data = torch.from_numpy(y_numpy).float().reshape(-1, 1)

# 2. 定义多层神经网络模型
class Model(torch.nn.Module):
    def __init__(self):
        super(Model, self).__init__()
        # 清理了多余的嵌套和缩进，正确注册网络层
        self.linear1 = torch.nn.Linear(10, 6)
        self.linear2 = torch.nn.Linear(6, 4)
        self.linear3 = torch.nn.Linear(4, 1)
        self.sigmoid = torch.nn.Sigmoid()

    def forward(self, x):
        x = self.sigmoid(self.linear1(x))
        x = self.sigmoid(self.linear2(x))
        x = self.sigmoid(self.linear3(x))
        return x

model = Model()

# 3. 损失函数与优化器
# ⚠️ 注意：由于 sklearn 糖尿病数据集是【回归任务】（预测连续的疾病指数，而不是 0 或 1 分类）
# 官方标准是用 MSELoss。如果你坚持用分类的 BCEWithLogitsLoss 会因目标值不是 0/1 而导致数学计算无意义。
# 这里为你修正为更符合数据集属性的 MSELosscriterion = torch.nn.MSELoss(reduction='sum')
optimizer = torch.optim.SGD(model.parameters(), lr=0.001) # 稍微调小学习率防止多层网络梯度爆炸

# 4. 训练循环
for epoch in range(100):
    y_pred = model(x_data)
    loss = criterion(y_pred, y_data)
    print(epoch, loss.item())

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
```

```
import numpy as np  
import torch  
from sklearn.datasets import load_diabetes  
from sklearn.preprocessing import StandardScaler  # 引入标准化工具  
  
# 1. 加载数据  
diabetes = load_diabetes()  
x_numpy = diabetes.data  
y_numpy = diabetes.target.reshape(-1, 1)  # 提前转成二维  
  
# 🌟 核心魔法：将 y 压缩到均值为 0，方差为 1 的正态分布  
scaler_y = StandardScaler()  
y_scaled = scaler_y.fit_transform(y_numpy)  
  
# 转换为 Tensorx_data = torch.from_numpy(x_numpy).float()  
y_data = torch.from_numpy(y_scaled).float()  # 使用标准化后的 y  
  
# 2. 定义非线性网络  
class RegressionModel(torch.nn.Module):  
    def __init__(self):  
        super(RegressionModel, self).__init__()  
        self.linear1 = torch.nn.Linear(10, 6)  
        self.linear2 = torch.nn.Linear(6, 4)  
        self.linear3 = torch.nn.Linear(4, 1)  
        self.relu = torch.nn.ReLU()  
  
    def forward(self, x):  
        x = self.relu(self.linear1(x))  
        x = self.relu(self.linear2(x))  
        x = self.linear3(x)  
        return x  
  
  
model = RegressionModel()  
  
# 3. 损失函数与优化器（调小一点学习率，配合标准化数据）  
criterion = torch.nn.MSELoss(reduction='mean')  
optimizer = torch.optim.Adam(model.parameters(), lr=0.01)  
  
# 4. 训练  
for epoch in range(100):  
    y_pred = model(x_data)  
    loss = criterion(y_pred, y_data)  
  
    if epoch % 10 == 0:  
        print(f"Epoch: {epoch}, Loss: {loss.item():.4f}")  
  
    optimizer.zero_grad()  
    loss.backward()  
    optimizer.step()  
  
# 5. 测试：如何把预测出来的“小数字”变回“真实的疾病指数”？  
print("\n--- 训练结束验证 ---")  
model.eval()  # 切换到评估模式  
with torch.no_grad():  
    # 拿第一个样本做预测  
    test_pred_scaled = model(x_data[0:1])  
    # 使用 inverse_transform 把结果反标准化  
    test_pred_real = scaler_y.inverse_transform(test_pred_scaled.numpy())  
  
    print(f"真实目标值: {y_numpy[0][0]}")  
    print(f"模型预测值: {test_pred_real[0][0]:.2f}")
```
