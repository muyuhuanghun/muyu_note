## 一、为什么需要模型保存与加载？

### 1. 训练中断的困境

```
📌 核心问题：训练可能随时中断

原因：
- 服务器宕机、断电
- 显存不足 (OOM)
- 训练时间太长，需要分多次进行
- 需要调整超参数重新训练

如果没有保存模型：
- 几小时/几天的训练成果全部丢失
- 必须从头开始训练
- 浪费大量时间和计算资源
```

### 2. 模型保存与加载的作用

```
💡 核心直觉：把训练好的模型"存档"，需要时"读档"

作用：
- 训练中断后恢复：从上次保存的地方继续训练
- 保存最佳模型：记录验证集上表现最好的模型
- 模型部署：把训练好的模型用于推理
- 迁移学习：把预训练模型迁移到新任务
```

## 二、PyTorch 模型保存与加载的原理

### 1. state_dict 的概念

```python
# 🌟 state_dict：模型参数的字典

import torch
import torch.nn as nn

# 定义一个简单模型
class SimpleModel(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(10, 20)
        self.fc2 = nn.Linear(20, 2)
    
    def forward(self, x):
        x = self.fc1(x)
        x = self.fc2(x)
        return x

model = SimpleModel()

# 查看 state_dict
print("模型的 state_dict:")
for key, value in model.state_dict().items():
    print(f"  {key}: {value.shape}")

# 输出：
# fc1.weight: torch.Size([20, 10])
# fc1.bias: torch.Size([20])
# fc2.weight: torch.Size([2, 20])
# fc2.bias: torch.Size([2])

# state_dict 是一个字典，包含所有可学习参数
# 不包含模型结构，只包含参数值
```

### 2. 为什么用 state_dict？

```
📌 state_dict 的优势：

1. 灵活性：
   - 可以只保存参数，不保存模型结构
   - 模型结构可以在代码中定义
   - 方便修改模型结构后加载旧参数

2. 兼容性：
   - 不同版本的 PyTorch 都能加载
   - 不依赖 Python 的 pickle 协议

3. 安全性：
   - 不会保存模型的计算图
   - 文件更小，加载更快
```

## 三、模型保存与加载的方法

### 1. 保存和加载 state_dict（推荐）

```python
# 🌟 推荐方法：只保存和加载 state_dict

# 保存
torch.save(model.state_dict(), 'model_weights.pth')

# 加载
model = SimpleModel()  # 先创建模型实例
model.load_state_dict(torch.load('model_weights.pth'))
model.eval()  # 设置为评估模式

# 优点：
# - 文件小（只有参数）
# - 灵活（可以加载到不同的模型结构）
# - 安全（不包含计算图）
```

### 2. 保存和加载整个模型

```python
# 🌟 另一种方法：保存整个模型

# 保存
torch.save(model, 'model.pth')

# 加载
model = torch.load('model.pth')
model.eval()

# 缺点：
# - 文件大（包含模型结构和参数）
# - 依赖 Python 环境和类定义
# - 不推荐用于生产环境
```

### 3. 保存和加载 Checkpoint（最完整）

```python
# 🌟 Checkpoint：保存训练的所有状态

# 保存 Checkpoint
checkpoint = {
    'epoch': epoch,
    'model_state_dict': model.state_dict(),
    'optimizer_state_dict': optimizer.state_dict(),
    'scheduler_state_dict': scheduler.state_dict(),
    'loss': loss,
    'best_val_loss': best_val_loss,
    'best_val_acc': best_val_acc,
}
torch.save(checkpoint, 'checkpoint.pth')

# 加载 Checkpoint
checkpoint = torch.load('checkpoint.pth')
model.load_state_dict(checkpoint['model_state_dict'])
optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
scheduler.load_state_dict(checkpoint['scheduler_state_dict'])
start_epoch = checkpoint['epoch']
best_val_loss = checkpoint['best_val_loss']

# 恢复训练
for epoch in range(start_epoch, num_epochs):
    train(...)
    validate(...)
```

## 四、最佳实践

### 1. 保存最佳模型

```python
# 🌟 训练过程中保存验证集上表现最好的模型

best_val_loss = float('inf')
best_model_path = 'best_model.pth'

for epoch in range(num_epochs):
    # 训练
    train_loss = train(...)
    
    # 验证
    val_loss, val_acc = validate(...)
    
    # 保存最佳模型
    if val_loss < best_val_loss:
        best_val_loss = val_loss
        torch.save(model.state_dict(), best_model_path)
        print(f"保存最佳模型，验证损失: {val_loss:.4f}")
    
    # 保存 Checkpoint（用于恢复训练）
    if (epoch + 1) % 10 == 0:  # 每 10 个 epoch 保存一次
        checkpoint = {
            'epoch': epoch + 1,
            'model_state_dict': model.state_dict(),
            'optimizer_state_dict': optimizer.state_dict(),
            'scheduler_state_dict': scheduler.state_dict(),
            'best_val_loss': best_val_loss,
        }
        torch.save(checkpoint, f'checkpoint_epoch_{epoch+1}.pth')
```

### 2. 定期保存 Checkpoint

```python
# 🌟 定期保存 Checkpoint，防止训练中断丢失太多进度

def save_checkpoint(model, optimizer, scheduler, epoch, loss, path):
    """保存 Checkpoint"""
    checkpoint = {
        'epoch': epoch,
        'model_state_dict': model.state_dict(),
        'optimizer_state_dict': optimizer.state_dict(),
        'scheduler_state_dict': scheduler.state_dict(),
        'loss': loss,
    }
    torch.save(checkpoint, path)

def load_checkpoint(model, optimizer, scheduler, path):
    """加载 Checkpoint"""
    checkpoint = torch.load(path)
    model.load_state_dict(checkpoint['model_state_dict'])
    optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
    scheduler.load_state_dict(checkpoint['scheduler_state_dict'])
    return checkpoint['epoch'], checkpoint['loss']

# 训练循环
for epoch in range(num_epochs):
    train_loss = train(...)
    
    # 每 5 个 epoch 保存一次
    if (epoch + 1) % 5 == 0:
        save_checkpoint(model, optimizer, scheduler, epoch, train_loss, 
                       f'checkpoint_epoch_{epoch+1}.pth')
    
    # 如果训练中断，可以这样恢复：
    # start_epoch, last_loss = load_checkpoint(model, optimizer, scheduler, 
    #                                         'checkpoint_epoch_10.pth')
```

### 3. 在 GPU 和 CPU 之间迁移

## 六、常见问题与解决方案

### 1. 加载时遇到 KeyError

```python
# 🌟 问题：模型结构改变后，加载旧的 state_dict

# 解决方案 1：strict=False（忽略不匹配的键）
model.load_state_dict(torch.load('model.pth'), strict=False)

# 解决方案 2：手动过滤匹配的键
def load_partial_state_dict(model, state_dict):
    """只加载匹配的参数"""
    model_state_dict = model.state_dict()
    
    # 过滤出匹配的参数
    filtered_state_dict = {
        k: v for k, v in state_dict.items() 
        if k in model_state_dict and v.shape == model_state_dict[k].shape
    }
    
    # 更新模型参数
    model_state_dict.update(filtered_state_dict)
    model.load_state_dict(model_state_dict)
    
    print(f"加载了 {len(filtered_state_dict)}/{len(model_state_dict)} 个参数")

load_partial_state_dict(model, torch.load('old_model.pth'))
```

### 2. 多 GPU 训练的模型保存

```python
# 🌟 问题：多 GPU 训练的模型如何保存和加载

# 多 GPU 训练时，模型被包装在 DataParallel 中
model = nn.DataParallel(model)

# 保存时，需要去掉 'module.' 前缀
torch.save(model.module.state_dict(), 'model.pth')

# 或者保存整个 DataParallel 模型
torch.save(model.state_dict(), 'model_dp.pth')

# 加载时：
# 如果要在单 GPU 上加载
model = SimpleModel()
state_dict = torch.load('model.pth')
# 如果保存的是 DataParallel 的 state_dict，需要去掉 'module.' 前缀
new_state_dict = {k.replace('module.', ''): v for k, v in state_dict.items()}
model.load_state_dict(new_state_dict)
```

## 七、对整体学习路线的影响

### 1. 在后续任务中的应用

| 任务 | 模型保存策略 |
|------|-------------|
| **UNet** | 保存最佳分割模型（基于 Dice Score） |
| **YOLO** | 保存最佳检测模型（基于 mAP） |
| **FaceNet** | 保存最佳特征提取器（基于 AUC） |
| **Transformer** | 保存 Checkpoint + 学习率调度器状态 |

### 2. 与迁移学习的关系

```
📌 迁移学习的基础：

1. 保存预训练模型的 state_dict
2. 在新任务中加载预训练模型
3. 冻结部分层，微调其他层
4. 保存微调后的模型

没有模型保存与加载，迁移学习无法实现
```

### 3. 在模型部署中的重要性

```
📌 模型部署流程：

1. 训练模型，保存最佳权重
2. 加载权重，转换为部署格式（ONNX、TorchScript）
3. 部署到服务器/边缘设备
4. 推理时加载模型权重

模型保存与加载是连接训练和部署的桥梁
```

## 八、代码验证（完整训练循环）

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.optim.lr_scheduler import StepLR

# 定义模型
class SimpleModel(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(784, 256)
        self.fc2 = nn.Linear(256, 10)
        self.relu = nn.ReLU()
    
    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.fc2(x)
        return x

# 初始化
model = SimpleModel()
optimizer = optim.Adam(model.parameters(), lr=1e-3)
scheduler = StepLR(optimizer, step_size=10, gamma=0.1)
criterion = nn.CrossEntropyLoss()

# 训练循环
best_val_acc = 0
for epoch in range(50):
    # 训练
    model.train()
    for images, labels in train_loader:
        images = images.view(-1, 784)
        outputs = model(images)
        loss = criterion(outputs, labels)
        
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
    
    scheduler.step()
    
    # 验证
    model.eval()
    correct = 0
    total = 0
    with torch.no_grad():
        for images, labels in val_loader:
            images = images.view(-1, 784)
            outputs = model(images)
            _, predicted = torch.max(outputs, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()
    
    val_acc = correct / total
    
    # 保存最佳模型
    if val_acc > best_val_acc:
        best_val_acc = val_acc
        torch.save(model.state_dict(), 'best_model.pth')
        print(f"保存最佳模型，验证准确率: {val_acc:.4f}")
    
    # 每 10 个 epoch 保存 Checkpoint
    if (epoch + 1) % 10 == 0:
        checkpoint = {
            'epoch': epoch + 1,
            'model_state_dict': model.state_dict(),
            'optimizer_state_dict': optimizer.state_dict(),
            'scheduler_state_dict': scheduler.state_dict(),
            'best_val_acc': best_val_acc,
        }
        torch.save(checkpoint, f'checkpoint_epoch_{epoch+1}.pth')
    
    print(f"Epoch {epoch+1}, Loss: {loss.item():.4f}, Val Acc: {val_acc:.4f}")
```
