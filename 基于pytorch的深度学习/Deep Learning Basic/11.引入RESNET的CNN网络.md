import torch  
from torchvision import transforms  
from torchvision import datasets  
from torch.utils.data import DataLoader  
import torch.nn.functional as F  
import torch.optim as optim  
  
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')  
print(f"🚀 Inception + ResNet 联合炼丹炉启动！当前设备：【{device}】")  
  
batch_size = 64  
  
transform = transforms.Compose([  
    transforms.ToTensor(),  
    transforms.Normalize((0.1307,), (0.3081,))  
])  
  
train_dataset = datasets.MNIST(root='../dataset/mnist/', train=True, download=True, transform=transform)  
train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)  
  
test_dataset = datasets.MNIST(root='../dataset/mnist/', train=False, download=True, transform=transform)  
test_loader = DataLoader(test_dataset, shuffle=False, batch_size=batch_size)  
  
  
# ==========================================  
# 1. 核心骨架 A：Inception 模块（保持不变）  
# ==========================================  
class InceptionA(torch.nn.Module):  
    def __init__(self, in_channels):  
        super(InceptionA, self).__init__()  
        self.branch1X = torch.nn.Conv2d(in_channels, 16, kernel_size=1)  
  
        self.branch5X_1 = torch.nn.Conv2d(in_channels, 16, kernel_size=1)  
        self.branch5X_2 = torch.nn.Conv2d(16, 24, kernel_size=5, padding=2)  
  
        self.branch3X_1 = torch.nn.Conv2d(in_channels, 16, kernel_size=1)  
        self.branch3X_2 = torch.nn.Conv2d(16, 24, kernel_size=3, padding=1)  
        self.branch3X_3 = torch.nn.Conv2d(24, 24, kernel_size=1)  
  
        self.branch_pool = torch.nn.Conv2d(in_channels, 24, kernel_size=1)  
  
    def forward(self, x):  
        branch1X = self.branch1X(x)  
  
        branch5X = self.branch5X_1(x)  
        branch5X = self.branch5X_2(branch5X)  
  
        branch3X = self.branch3X_1(x)  
        branch3X = self.branch3X_2(branch3X)  
        branch3X = self.branch3X_3(branch3X)  
  
        branch_pool = F.avg_pool2d(x, kernel_size=3, stride=1, padding=1)  
        branch_pool = self.branch_pool(branch_pool)  
  
        outputs = [branch1X, branch5X, branch3X, branch_pool]  
        return torch.cat(outputs, dim=1)  
  
  
# ==========================================  
# 📌 2. 核心骨架 B：手搓 ResNet 残差块  
# ==========================================  
class ResidualBlock(torch.nn.Module):  
    def __init__(self, channels):  
        super(ResidualBlock, self).__init__()  
        # 保持通道数和宽高不变的经典两层卷积结构（padding=1配合kernel_size=3）  
        self.conv1 = torch.nn.Conv2d(channels, channels, kernel_size=3, padding=1)  
        self.conv2 = torch.nn.Conv2d(channels, channels, kernel_size=3, padding=1)  
  
    def forward(self, x):  
        # 1. 走主路 F(x)        y = F.relu(self.conv1(x))  
        y = self.conv2(y)  
        # 2. 抄近路，将主路输出和原始输入 x 强行相加，然后再过一遍激活函数  
        return F.relu(y + x)  
  
  
# ==========================================  
# 3. 终极主网络（融合 Inception + ResNet）  
# ==========================================  
class Net(torch.nn.Module):  
    def __init__(self):  
        super(Net, self).__init__()  
        self.conv1 = torch.nn.Conv2d(1, 10, kernel_size=5)  
        self.conv2 = torch.nn.Conv2d(88, 20, kernel_size=5)  
  
        self.incep1 = InceptionA(10)  # 输出 88 通道  
        self.incep2 = InceptionA(20)  # 输出 88 通道  
  
        # 📌 引入残差块：分别接在 Inception 模块后面  
        # 必须传入对应的通道数（第一个Inception输出88通道，第二个输出88通道）  
        self.res1 = ResidualBlock(88)  
        self.res2 = ResidualBlock(88)  
  
        self.mp = torch.nn.MaxPool2d(2)  
        # 📌 尺寸推导重新验证：  
        # 最后的incep2+res2输出依然是 88通道，宽高依然是 4x4        # 88 * 4 * 4 = 1408 维，完全不需要修改全连接层尺寸！  
        self.fc = torch.nn.Linear(1408, 10)  
  
    def forward(self, x):  
        in_size = x.size(0)  
  
        # 第一阶段  
        x = F.relu(self.mp(self.conv1(x)))  
        x = self.incep1(x)  
        x = self.res1(x)  # 📌 激活残差连接 1  
        # 第二阶段  
        x = F.relu(self.mp(self.conv2(x)))  
        x = self.incep2(x)  
        x = self.res2(x)  # 📌 激活残差连接 2  
        # 展平送入分类器  
        x = x.view(in_size, -1)  
        x = self.fc(x)  
        return x  
  
  
model = Net().to(device)  
  
criterion = torch.nn.CrossEntropyLoss()  
optimizer = optim.SGD(model.parameters(), lr=0.001, momentum=0.9)  
  
  
def train(epoch):  
    running_loss = 0.0  
    for batch_idx, data in enumerate(train_loader, 0):  
        inputs, target = data  
        inputs, target = inputs.to(device), target.to(device)  
  
        optimizer.zero_grad()  
        outputs = model(inputs)  
        loss = criterion(outputs, target)  
        loss.backward()  
        optimizer.step()  
  
        running_loss += loss.item()  
        if batch_idx % 300 == 299:  
            print('[Epoch %d, Batch %5d] loss: %.3f' % (epoch + 1, batch_idx + 1, running_loss / 300))  
            running_loss = 0.0  
  
  
def test():  
    correct = 0  
    total = 0  
    with torch.no_grad():  
        for data in test_loader:  
            images, labels = data  
            images, labels = images.to(device), labels.to(device)  
  
            outputs = model(images)  
            _, predicted = torch.max(outputs.data, dim=1)  
            total += labels.size(0)  
            correct += (predicted == labels).sum().item()  
    print('🎯 Accuracy on test set: %d %%' % (100 * correct / total))  
    print("-" * 40)  
  
  
if __name__ == '__main__':  
    for epoch in range(10):  
        train(epoch)  
        test()
