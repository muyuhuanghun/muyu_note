```
import torch  
import numpy as np  
import matplotlib.pyplot as plt  
  
  
x_data = torch.tensor([[1.0], [2.0], [3.0]], dtype=torch.float32)  
y_data = torch.tensor([[0.0], [0.0], [1.0]], dtype=torch.float32)  # prepare dataset  
  
#--------------------------------------------#  
  
class LogisticRegressionModel(torch.nn.Module):  
    def __init__(self):  
        super(LogisticRegressionModel, self).__init__()  
        self.linear = torch.nn.Linear(1, 1)  
  
    def forward(self, x):  
        return self.linear(x)  
  
model = LogisticRegressionModel()             #Design Model Using Class  
  
#---------------------------------------------#  
  
criterion = torch.nn.BCEWithLogitsLoss(reduction='sum')  
optimizer = torch.optim.SGD(model.parameters(), lr=0.01)  #construct loss and optimizer  
  
#--------------------------------------------#  
  
model.train()  
for epoch in range(1000):  
    y_pred = model(x_data)  
    loss = criterion(y_pred, y_data)  
    print(epoch, loss.item())  
  
    optimizer.zero_grad()  
    loss.backward()  
    optimizer.step()                        #training cycle  
  
x = np.linspace(0, 10, 200, dtype=np.float32)  
x_t = torch.from_numpy(x).view(200, 1)  
model.eval()  
with torch.no_grad():  
    y_t = torch.sigmoid(model(x_t))  
y = y_t.detach().numpy()  
plt.plot(x, y)  
plt.plot([0, 10], [0.5, 0.5], c='r')  
plt.xlabel('hours')  
plt.ylabel('probability')  
plt.grid()  
plt.show()
```
