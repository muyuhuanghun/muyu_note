import numpy as np  
import torch  
from torch.utils.data import Dataset, DataLoader  
  
# 🌟 【新加】步骤一：检测你的电脑有没有 NVIDIA 显卡（CUDA环境）  
# 如果有，就用 GPU ('cuda')；如果没有，就退回 CPUdevice = torch.device('cuda' if torch.cuda.is_available() else 'cpu')  
print(f"当前训练使用的设备是: {device}")  
  
  
class DiabetesDataset(Dataset):  
    def __init__(self, filepath):  
        xy = np.loadtxt(filepath, delimiter=' ', dtype=np.float32)  
        self.len = xy.shape[0]  
        self.x_data = torch.from_numpy(xy[:, :-1])  
        self.y_data = torch.from_numpy(xy[:, [-1]])  
  
    def __getitem__(self, index):  
        return self.x_data[index], self.y_data[index]  
  
    def __len__(self):  
        return self.len  
  
  
dataset = DiabetesDataset(r'E:\deep learning\diabetes_data_raw.csv.gz')  
# 开启 persistent_workers=True 驻留工人，配合 GPU 压榨速度  
train_loader = DataLoader(dataset=dataset, batch_size=32, shuffle=True, num_workers=2, persistent_workers=True)  
  
  
class Model(torch.nn.Module):  
    def __init__(self):  
        super(Model, self).__init__()  
        self.linear1 = torch.nn.Linear(9, 12)  
        self.linear2 = torch.nn.Linear(12, 6)  
        self.linear3 = torch.nn.Linear(6, 1)  
        self.sigmoid = torch.nn.Sigmoid()  
  
    def forward(self, x):  
        x = self.sigmoid(self.linear1(x))  
        x = self.sigmoid(self.linear2(x))  
        x = self.linear3(x)  
        return x  
  
  
model = Model()  
# 🌟 【新加】步骤二：把模型的所有权重和偏置，一键传送进显存  
model = model.to(device)  
  
criterion = torch.nn.MSELoss(reduction='mean')  
optimizer = torch.optim.SGD(model.parameters(), lr=0.01)  
  
if __name__ == '__main__':  
    for epoch in range(100):  
        for i, data in enumerate(train_loader, 0):  
            inputs, labels = data  
  
            # 🌟 【新加】步骤三：把从 DataLoader 运出来的这批小数据，也传送进显存  
            # 💡 注意：模型和数据必须在同一个设备（都在显存或都在内存），否则会报错  
            inputs = inputs.to(device)  
            labels = labels.to(device)  
  
            # 下面的前向传播、计算损耗、反向传播，就会全自动在显存里以极速狂飙  
            y_pred = model(inputs)  
            loss = criterion(y_pred, labels)  
  
            print(epoch, i, loss.item())  
  
            optimizer.zero_grad()  
            loss.backward()  
            optimizer.step()
