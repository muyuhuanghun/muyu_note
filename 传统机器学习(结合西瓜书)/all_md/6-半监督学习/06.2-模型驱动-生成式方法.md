## 一、半监督生成式方法

> 如何用概率模型同时利用有标签和无标签数据进行分类？

生成式方法: 假设所有数据(无论是否有标记)都是由同一个潜在的模型"生成"的,未标记数据的标记可以视为模型缺失的参数,一般可基于EM算法进行极大似然估计求解(假设的生成模型必须与真实的数据分布相互吻合)

```python
import numpy as np
from sklearn.mixture import GaussianMixture
from sklearn.datasets import make_blobs
from sklearn.metrics import accuracy_score

# 📌 核心思想：用GMM拟合全部数据(有标+无标)的分布形状
# 然后用少量有标签数据给每个高斯成分"命名"
# 就像你先用所有学生(不管成绩已知未知)的成绩分布来划分"学霸/中等/后进"三个档次
# 然后用少数已知成绩的学生来确定哪个档次对应哪个分数段

np.random.seed(42)

# ---- 1. 生成数据：两个类别，有重叠 ----
X, y = make_blobs(n_samples=500, centers=2, n_features=2,
                   cluster_std=1.5, random_state=42)

# ---- 2. 模拟半监督场景：只保留20个标签 ----
rng = np.random.RandomState(42)
n_labeled = 20
mask_unlabeled = rng.rand(len(y)) > (n_labeled / len(y))
y_semi = y.copy()
y_semi[mask_unlabeled] = -1  # -1 表示无标签

# ---- 3. 用GMM拟合全部数据的分布（Step 1：无监督的聚类）----
# 这是生成式方法的关键：用所有数据(不管有没有标签)来训练GMM
# GMM 会自己发现数据天然分成了几个"簇"
n_components = 2  # 假设我们知道有两个类别
gmm = GaussianMixture(n_components=n_components, covariance_type='full',
                       random_state=42)
gmm.fit(X)  # ⚠️ 这里用了全部数据，包括无标签的！

# 得到每个样本属于每个高斯成分的概率（responsibilities）
responsibilities = gmm.predict_proba(X)  # shape: (500, 2)
# responsibilities[i, j] = 样本i由第j个高斯成分"生成"的概率

# ---- 4. 用有标签数据给每个高斯成分"命名"（Step 2：成分→类别映射）----
# 少数有标签样本告诉我们：哪个高斯簇对应哪个真实类别
labeled_idx = np.where(~mask_unlabeled)[0]  # 有标签样本的索引

# 统计每个高斯成分中，各类别的有标签样本数量
# component_class[i, j] = 第i个成分中有多少个第j类的有标签样本
component_class = np.zeros((n_components, n_components))
for idx in labeled_idx:
    best_component = np.argmax(responsibilities[idx])  # 该样本最可能属于哪个成分
    true_class = y[idx]
    component_class[best_component, true_class] += 1

# 为每个成分分配"票数最多"的类别
component_to_class = np.argmax(component_class, axis=1)
print("高斯成分 → 类别映射:", component_to_class)
# 输出示例: [0 1] 表示成分0 → 类别0，成分1 → 类别1

# ---- 5. 对新样本预测（Step 3：看它落入哪个已命名的"簇"）----
def gmm_semi_predict(X_new, gmm, component_to_class):
    """半监督生成式方法的预测函数"""
    resp = gmm.predict_proba(X_new)        # 新样本属于每个成分的概率
    pred_components = np.argmax(resp, axis=1)  # 最可能的高斯成分
    return component_to_class[pred_components]  # 映射到真实类别

y_pred = gmm_semi_predict(X, gmm, component_to_class)
acc = accuracy_score(y, y_pred)
print(f"半监督生成式方法 (仅{n_labeled}个标签) 准确率: {acc:.4f}")
# 💡 仅用 20/500 (4%) 个标签就达到了不错的准确率
# 因为无标签数据帮你摸清了数据的分布形状
```

```python
# ============================================
# 🌟 EM算法的核心逻辑 — 从零实现
# 上面用的是 sklearn 的 GaussianMixture（内部已实现EM）
# 但理解EM的"软硬贡献"机制，才能看懂西瓜书公式(13.6)-(13.8)
# ============================================

import numpy as np
from scipy.stats import multivariate_normal

class SemiSupervisedGMM:
    """半监督高斯混合模型 — 手写实现，演示EM核心机制"""

    def __init__(self, n_components=2, max_iter=100, tol=1e-4):
        self.n_components = n_components
        self.max_iter = max_iter
        self.tol = tol

    def fit(self, X, y_semi):
        """
        X: 全部样本 (N, D)
        y_semi: 半监督标签，-1表示无标记，0/1等表示类别
        """
        N, D = X.shape
        K = self.n_components
        y_semi = y_semi.astype(int)

        labeled_mask = y_semi != -1
        unlabeled_mask = ~labeled_mask

        # ---- 初始化 GMM 参数 ----
        self.alpha = np.ones(K) / K                                    # 混合系数
        self.mu = X[np.random.choice(N, K, replace=False)]            # 均值（随机选K个样本做初始中心）
        self.sigma = np.array([np.eye(D) for _ in range(K)])          # 协方差矩阵

        log_likelihood_old = -np.inf
        for iteration in range(self.max_iter):
            # ============================================
            # E-step: 计算每个样本属于每个成分的后验概率
            # 这就是式13.3的"响应度"(responsibility)
            # ============================================
            gamma = np.zeros((N, K))  # gamma[n, k] = 样本n属于成分k的概率

            for k in range(K):
                # 用当前参数计算每个样本在第k个高斯分布下的概率密度
                gamma[:, k] = self.alpha[k] * multivariate_normal.pdf(
                    X, mean=self.mu[k], cov=self.sigma[k]
                )
            gamma /= gamma.sum(axis=1, keepdims=True)  # 归一化

            # ============================================
            # M-step: 更新参数
            # 有标签数据给出"硬贡献"(确定性的)，无标签数据给出"软贡献"(按概率)
            # 这就是式13.6~13.8的核心逻辑
            # ============================================
            for k in range(K):
                # --- 硬贡献：有标签且标签==k 的样本，贡献权重=1 ---
                hard_mask = (y_semi == k)  # 注意：只有有标签且类别恰好是k的
                n_hard = hard_mask.sum()

                # --- 软贡献：无标签样本按 gamma 分配 ---
                soft_weights = gamma[unlabeled_mask, k]  # 无标签样本对成分k的贡献
                total_soft = soft_weights.sum()

                # 🌟 总有效样本数 = 硬标签的个数 + 软标签的概率和（式13.8的分母逻辑）
                total_weight = n_hard + total_soft

                if total_weight > 0:
                    # --- 均值更新(式13.6)：找成分k的中心 ---
                    # 分子 = 无标签样本的期望位置 + 有标签样本的实际位置
                    mu_numerator = (soft_weights @ X[unlabeled_mask] +
                                    X[hard_mask].sum(axis=0))
                    self.mu[k] = mu_numerator / total_weight

                    # --- 协方差更新(式13.7)：看成分k内部的散布 ---
                    X_centered_all = X - self.mu[k]  # 所有样本相对于中心的偏离
                    sigma_numerator = np.zeros((D, D))
                    # 无标签部分的软加权协方差
                    for i, idx in enumerate(np.where(unlabeled_mask)[0]):
                        diff = X_centered_all[idx].reshape(-1, 1)
                        sigma_numerator += soft_weights[i] * (diff @ diff.T)
                    # 有标签部分的硬加权协方差
                    for idx in np.where(hard_mask)[0]:
                        diff = X_centered_all[idx].reshape(-1, 1)
                        sigma_numerator += diff @ diff.T
                    self.sigma[k] = sigma_numerator / total_weight

                    # --- 混合系数更新(式13.8)：算第k类在"全校"的占比 ---
                    self.alpha[k] = total_weight / N

            # ---- 收敛判断 ----
            log_likelihood = self._compute_log_likelihood(X)
            if np.abs(log_likelihood - log_likelihood_old) < self.tol:
                print(f"EM在第{iteration+1}轮收敛")
                break
            log_likelihood_old = log_likelihood

    def _compute_log_likelihood(self, X):
        """计算对数似然，用于判断收敛"""
        log_lik = 0
        for k in range(self.n_components):
            component_lik = self.alpha[k] * multivariate_normal.pdf(
                X, mean=self.mu[k], cov=self.sigma[k]
            )
            log_lik += np.sum(np.log(np.maximum(component_lik, 1e-15)))
        return log_lik

    def predict(self, X):
        """预测：计算 p(y|X) = sum over components p(y|component) * p(component|X)"""
        gamma = np.zeros((X.shape[0], self.n_components))
        for k in range(self.n_components):
            gamma[:, k] = self.alpha[k] * multivariate_normal.pdf(
                X, mean=self.mu[k], cov=self.sigma[k]
            )
        gamma /= gamma.sum(axis=1, keepdims=True)
        return np.argmax(gamma, axis=1)  # 取后验概率最大的成分 = 类别
```

```python
# ============================================
# 📌 验证手写半监督GMM
# ============================================

sgmm = SemiSupervisedGMM(n_components=2, max_iter=100)
sgmm.fit(X, y_semi)
y_pred_hand = sgmm.predict(X)
acc_hand = accuracy_score(y, y_pred_hand)
print(f"手写半监督GMM准确率: {acc_hand:.4f}")
print(f"sklearn GMM准确率:    {acc:.4f}")
print("💡 两者效果接近，验证了EM算法中'硬贡献+软贡献'框架的正确性")
```

## 二、为什么无标签数据能提升效果？

> 无标签数据到底在半监督学习中扮演什么角色？

```python
# 🌟 用大白话理解：
# 有标签数据的任务是："点名" — 告诉我这个高斯簇叫"猫"，那个叫"狗"
# 无标签数据的任务是："画地图" — 告诉模型数据长什么样、有几个团块
#
# 你只需要3张图上标好"这是猫"、"这是狗"
# 剩下997张无标签图片虽然没写答案，但它们帮模型精确地画出了
# "猫类图片"的分布范围和"狗类图片"的分布范围
# 这样模型就知道分界线该画在哪了

# 📌 实验：对比不同标签数量下的效果
def compare_label_counts(X, y, label_counts=[5, 10, 20, 50, 100]):
    """对比不同标注数量下半监督GMM的表现"""
    for n_labeled in label_counts:
        rng = np.random.RandomState(42)
        mask = rng.rand(len(y)) > (n_labeled / len(y))
        y_semi = y.copy()
        y_semi[mask] = -1

        gmm = GaussianMixture(n_components=2, covariance_type='full', random_state=42)
        gmm.fit(X)

        # 用有标签样本建立成分→类别映射
        responsibilities = gmm.predict_proba(X)
        labeled_idx = np.where(~mask)[0]

        component_class = np.zeros((2, 2))
        for idx in labeled_idx:
            best_comp = np.argmax(responsibilities[idx])
            component_class[best_comp, y[idx]] += 1
        comp_to_class = np.argmax(component_class, axis=1)

        y_pred = comp_to_class[np.argmax(responsibilities, axis=1)]
        acc = accuracy_score(y, y_pred)
        print(f"  标签数={n_labeled:3d} ({n_labeled/len(y)*100:5.1f}%) → 准确率={acc:.4f}")

print("📌 标签数量对半监督GMM的影响:")
print("-" * 50)
compare_label_counts(X, y)
# 💡 观察：即使标签从50降到10，准确率下降也很有限
# 说明无标签数据承担的"画地图"任务才是决定性能的关键
```

```python
# 📌 式13.2 的本质：把分类拆成两步
# p(y=j | x) = sum_i p(y=j | component=i, x) * p(component=i | x)
#    ^              ^                          ^
#   分类结果      哪个成分=哪个类            样本是哪个成分的概率
#                (有标签数据来"点名")       (无标签数据来"画地图")

# 通俗比喻：
# 你是学校教务处的，要把新生分到"尖子班"还是"普通班"
# 第一步：通过所有新生的入学成绩分布，发现分成"高分群"和"低分群"（无监督）
# 第二步：少数几个已确定分班的学生，告诉你"高分群"就是尖子班（有监督）
# 第三步：新来的学生看成绩落在哪个群，就分到哪个班（预测）
```

## 四、EM公式的"软硬贡献"结构

> EM算法如何平衡有标签和无标签数据的贡献？

```python
# 📌 式13.6-13.8的分子都是 "左边一个Σ + 右边一个Σ"，解读如下：
# 
# 右边的Σ（硬贡献）= 来自有标签数据
#   → 这些样本的身份确定，如果标签说它是第i类，
#     那它对第i类中心的贡献就是 1 × x_j（实打实的1票）
# 
# 左边的Σ（软贡献）= 来自无标签数据
#   → 这些样本身份不明，按概率γ_ji分配"贡献额度"
#     如果一个样本只有30%概率属于第i类，
#     那它对第i类中心的影响力就是 0.3 × x_j（打了折扣的0.3票）
# 
# 分母（总有效样本数）= 硬标签个数 + 软标签概率和
#   → 统计第i类一共"招募"了多少有效影响力

# 🌟 这种设计的精妙之处：
# 如果你有很多有标签数据 → l_i很大，公式听标签的指挥
# 如果你只有极少标签但海量无标签 → Σγ_ji主导，模型更多挖掘数据本身的分布规律
```
