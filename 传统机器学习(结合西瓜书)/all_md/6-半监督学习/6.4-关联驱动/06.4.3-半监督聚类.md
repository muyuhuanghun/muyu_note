## 一、半监督聚类 — 两种监督信息

半监督聚类中可以利用哪些监督信息来引导聚类过程？

### 1. 可以用的两种监督信息

- 第一种类型是 "必连"(must-link) 与 "勿连"(cannot-link) 约束，前者是指样本必属于同一个簇，后者是指样本必不属于同一个簇；
- 第二种类型的监督信息是少量的有标记样本。

```python
# 🌟 概念比喻：
# 必连/勿连约束：
#   老师说："张三和李四肯定在一个组" (must-link)
#   老师说："王五和赵六绝对不能在一个组" (cannot-link)
#   聚类算法在这些约束下，重新调整分组
# 
# 少量有标记样本：
#   老师说："这3个人是A组，那2个人是B组"
#   聚类算法以这些人为"种子"向外扩展分组

import numpy as np
from sklearn.datasets import make_blobs
from sklearn.cluster import KMeans
from sklearn.metrics import adjusted_rand_score, normalized_mutual_info_score
from scipy.spatial.distance import cdist
```

## 二、约束K均值算法

如何利用 must-link 和 cannot-link 约束来改进 K-means 聚类？

![alt text](image-6.png)
![alt text](image-7.png)

```python
class ConstrainedKMeans:
    """
    约束K均值：在标准K-means的基础上加入 must-link 和 cannot-link 约束

    核心修改在"样本分配"步骤：
    标准K-means: 把样本 x_i 分配给最近的簇中心
    约束版: 把样本 x_i 分配给最近的"合法"簇中心
            "合法" = 不会违反 must-link 和 cannot-link 约束

    比喻：
    教室里分组做项目
    标准分组：每个人去自己朋友最多的组
    约束分组：加上"张三李四必须在同组"(must-link)
              和"王五赵六必须不同组"(cannot-link)
    """

    def __init__(self, n_clusters=3, max_iter=100, tol=1e-4):
        self.n_clusters = n_clusters
        self.max_iter = max_iter
        self.tol = tol

    def fit(self, X, must_link_pairs=None, cannot_link_pairs=None):
        """
        X: 数据矩阵 (n, d)
        must_link_pairs: 必连对，如 [(0, 5), (3, 7)]  表示索引0和5必须在同一簇
        cannot_link_pairs: 勿连对，如 [(1, 4)] 表示索引1和4必须不同簇
        """
        X = np.array(X)
        n, d = X.shape

        if must_link_pairs is None:
            must_link_pairs = []
        if cannot_link_pairs is None:
            cannot_link_pairs = []

        # ---- 初始化簇中心 ----
        rng = np.random.RandomState(42)
        self.cluster_centers_ = X[rng.choice(n, self.n_clusters, replace=False)]

        # ---- 构建约束邻居关系（用于快速检查合法性）----
        # 🌟 must-link 是可传递的：如果 A-B 必连、B-C 必连，则 A-C 也必连
        # 这里用传递闭包把所有间接 must-link 也找出来
        self.must_link_sets = self._transitive_closure(must_link_pairs, n)
        self.cannot_link_map = self._build_cannot_link_map(cannot_link_pairs, n)

        labels = np.zeros(n, dtype=int)
        for iteration in range(self.max_iter):
            labels_old = labels.copy()

            # ---- 分配步骤：给每个样本找"合法"的最近簇 ----
            for i in range(n):
                distances = cdist(X[i:i+1], self.cluster_centers_).ravel()

                # 🌟 关键：按距离排序，找第一个"合法"的簇
                # 合法 = 不会让 must-link 伙伴分到不同簇
                #       不会让 cannot-link 对手分到同簇
                available = set(range(self.n_clusters))
                constrained = set()

                # 检查 must-link 约束
                for j in self.must_link_sets[i]:
                    if labels[j] >= 0:  # 如果 must-link 伙伴已被分配
                        constrained.add(labels[j])  # 必须去同一个簇

                # 检查 cannot-link 约束
                for j in self.cannot_link_map.get(i, []):
                    constrained = constrained.union(
                        {c for c in range(self.n_clusters) if c != labels[j]}
                    )
                    # ⚠️ 简化处理：实际算法需要更复杂的回溯

                # 如果有约束，只能在允许的簇中选择
                if len(constrained) > 0:
                    # 合并约束：必须去某个特定簇
                    valid_clusters = list(constrained)
                else:
                    valid_clusters = list(range(self.n_clusters))

                # 在合法簇中选择最近的
                valid_distances = distances[valid_clusters]
                best_local = valid_clusters[np.argmin(valid_distances)]
                labels[i] = best_local

            # ---- 更新步骤：重新计算簇中心 ----
            for k in range(self.n_clusters):
                members = X[labels == k]
                if len(members) > 0:
                    self.cluster_centers_[k] = members.mean(axis=0)

            # ---- 收敛判断 ----
            n_changed = np.sum(labels != labels_old)
            if n_changed < self.tol * n:
                print(f"约束K-means在第{iteration+1}轮收敛")
                break

        self.labels_ = labels
        return self

    def _transitive_closure(self, pairs, n):
        """计算 must-link 的传递闭包 — 用并查集合并所有间接必连"""
        parent = list(range(n))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a, b):
            ra, rb = find(a), find(b)
            if ra != rb:
                parent[ra] = rb

        for a, b in pairs:
            union(a, b)

        # 构建每个节点所在的 must-link 集合
        sets = {i: set() for i in range(n)}
        for i in range(n):
            root = find(i)
            for j in range(n):
                if find(j) == root and j != i:
                    sets[i].add(j)

        return sets

    def _build_cannot_link_map(self, pairs, n):
        """构建 cannot-link 映射"""
        cl_map = {}
        for a, b in pairs:
            cl_map.setdefault(a, set()).add(b)
            cl_map.setdefault(b, set()).add(a)
        return cl_map

    def predict(self, X):
        distances = cdist(X, self.cluster_centers_)
        return np.argmin(distances, axis=1)
```

```python
# ============================================
# 📌 测试约束K-means
# ============================================

# 生成团块数据
X_cluster, y_true = make_blobs(n_samples=300, centers=3, n_features=2,
                                random_state=42, cluster_std=1.5)

# 标准K-means（无约束）
kmeans_standard = KMeans(n_clusters=3, random_state=42, n_init=10)
y_standard = kmeans_standard.fit_predict(X_cluster)

# 约束K-means：加入 must-link 和 cannot-link 约束
# 假设我们知道少量成对关系：
must_links = [(0, 10), (5, 20), (100, 150)]   # 这些对必须在同一簇
cannot_links = [(0, 200), (10, 250)]            # 这些对必须不同簇

ckmeans = ConstrainedKMeans(n_clusters=3, max_iter=50)
ckmeans.fit(X_cluster, must_link_pairs=must_links, cannot_link_pairs=cannot_links)
y_constrained = ckmeans.labels_

print("📌 标准K-means vs 约束K-means:")
print(f"  标准K-means  ARI: {adjusted_rand_score(y_true, y_standard):.4f}")
print(f"  约束K-means  ARI: {adjusted_rand_score(y_true, y_constrained):.4f}")
print(f"💡 即使少量的成对约束，也能引导聚类朝向正确的方向")
```

## 三、约束种子K均值算法

如何利用少量有标签样本来引导聚类？

![alt text](image-8.png)

```python
class SeededKMeans:
    """
    种子K均值：用少量有标签样本作为"种子"，初始化并引导聚类

    核心机制：
    1. 用有标签样本作为"置信锚点"初始化簇中心
    2. 在迭代中确保有标签样本始终留在正确的簇中
    3. 无标签样本被"吸引"到最近的种子周围

    比喻：
    已知5个学生是"尖子班"(有标签)，8个是"普通班"
    以他们的平均成绩为"种子分数"
    其余学生的成绩向最接近的"种子分数"靠拢
    分类边界自然形成
    """

    def __init__(self, n_clusters=3, max_iter=100, tol=1e-4):
        self.n_clusters = n_clusters
        self.max_iter = max_iter
        self.tol = tol

    def fit(self, X, y_semi):
        """
        X: 全部样本 (n, d)
        y_semi: 半监督标签，-1=未标记，0/1/...=类别
        """
        X = np.array(X)
        y_semi = np.array(y_semi).astype(int)
        n, d = X.shape

        labeled_mask = y_semi != -1
        unlabeled_mask = y_semi == -1

        # 🌟 Step 1: 用有标签样本初始化簇中心（种子）
        self.cluster_centers_ = np.zeros((self.n_clusters, d))
        for k in range(self.n_clusters):
            members = X[(y_semi == k) & labeled_mask]
            if len(members) > 0:
                self.cluster_centers_[k] = members.mean(axis=0)
            else:
                # 如果某个类别没有种子，随机初始化
                self.cluster_centers_[k] = X[np.random.choice(n)]

        labels = y_semi.copy()
        # ⚠️ 有标签样本的标签始终锁定（这是和约束K-means的区别）

        for iteration in range(self.max_iter):
            labels_old = labels.copy()

            # ---- 只对无标签样本重新分配 ----
            X_unlabeled = X[unlabeled_mask]
            if len(X_unlabeled) == 0:
                break

            distances = cdist(X_unlabeled, self.cluster_centers_)
            # 分配到最近的簇
            new_labels = np.argmin(distances, axis=1)
            labels[unlabeled_mask] = new_labels

            # ---- 更新簇中心（所有样本，包括有标签的）----
            for k in range(self.n_clusters):
                members = X[labels == k]
                if len(members) > 0:
                    self.cluster_centers_[k] = members.mean(axis=0)

            # ---- 收敛判断 ----
            n_changed = np.sum(labels[unlabeled_mask] != labels_old[unlabeled_mask])
            if n_changed < self.tol * n:
                print(f"种子K-means在第{iteration+1}轮收敛")
                break

        self.labels_ = labels
        return self

    def predict(self, X):
        distances = cdist(X, self.cluster_centers_)
        return np.argmin(distances, axis=1)


# ============================================
# 📌 测试种子K-means
# ============================================

# 模拟半监督场景：只有5个标签
n_labeled = 5
rng = np.random.RandomState(42)
mask_unlabeled = rng.rand(len(y_true)) > (n_labeled / len(y_true))
y_semi_cluster = y_true.copy()
y_semi_cluster[mask_unlabeled] = -1

# 标准K-means
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
y_kmeans = kmeans.fit_predict(X_cluster)

# 种子K-means
skmeans = SeededKMeans(n_clusters=3, max_iter=50)
skmeans.fit(X_cluster, y_semi_cluster)
y_seeded = skmeans.labels_

print(f"\n📌 标准K-means vs 种子K-means:")
print(f"  标准K-means     ARI: {adjusted_rand_score(y_true, y_kmeans):.4f}")
print(f"  种子K-means     ARI: {adjusted_rand_score(y_true, y_seeded):.4f}")
print(f"💡 仅用 {n_labeled}/{len(y_true)} 个种子标签就显著改善了聚类质量")
```

```python
# ============================================
# 📌 对比：三种K-means变体的差异
# ============================================

# 标准K-means:
#   初始化：随机选K个点作中心
#   分配：  每个点分配给最近中心
#   更新：  重新计算中心
#   约束：  无

# 约束K-means:
#   初始化：随机选K个点作中心（同标准）
#   分配：  每个点分配给最近"合法"中心
#   更新：  重新计算中心（同标准）
#   约束：  must-link + cannot-link 成对约束
#   🌟 约束影响的是"分配"步骤

# 种子K-means:
#   初始化：用有标签样本的平均值作中心 🌟 引用了监督信息
#   分配：  无标签点分配给最近中心，有标签点锁定不变
#   更新：  重新计算中心（同标准）
#   约束：  少量有标记样本（种子）
#   🌟 监督信息影响的是"初始化"+锁定有标签样本

print("\n📌 三种K-means变体的关键区别:")
print("  1. 标准K-means:     零监督、纯无监督聚类")
print("  2. 约束K-means:     利用成对约束(must-link/cannot-link)")
print("  3. 种子K-means:     利用少量标签作为'锚点'初始化和引导")
```

## 四、半监督聚类的实用场景

```python
# 🌟 实际应用场景：
# 
# 1. 图像分割：
#    用户在图上点几个点："这片是天空"、"这片是草地"
#    半监督聚类把剩下的像素自动分到最相似的标记区域
#    → 约束K-means 或种子K-means
# 
# 2. 客户分群：
#    CRM中已知某几个客户是"高价值"、"中等价值"、"低价值"
#    以此为种子，自动给其余所有客户分群
#    → 种子K-means
# 
# 3. 文本聚类：
#    已知少数文档的类别，聚类算法以此为"锚点"整理所有文档
#    → 种子K-means
# 
# 4. 社交网络社区发现：
#    已知少数用户属于某个社区（如已知同事关系）
#    以此约束发现完整的社区结构
#    → 约束K-means (用must-link模拟已知同社区关系)
```
