## 一、半监督SVM核心概念

> 如何让SVM利用未标注数据找到更好的决策边界？

不考虑未标记样本时，SVM试图找到最大间隔划分的超平面；考虑后，S3VM试图找到能将两类标记样本有效分开，**且穿过数据低密度区域** 的划分超平面

```python
import numpy as np
from sklearn.svm import SVC
from sklearn.datasets import make_classification
from sklearn.metrics import accuracy_score, classification_report
import warnings
warnings.filterwarnings('ignore')

np.random.seed(42)

# 📌 TSVM的核心思想：
# 对未标记样本进行"各种可能的标记指派"
# 即把未标记样本分别假设成正类或反类
# 看看哪种假设能让分界线最宽、最稳
# 同时在聚类假设约束下，分界线应穿过数据密度最低的地方

# ---- 1. 生成数据并模拟半监督场景 ----
X, y = make_classification(n_samples=300, n_features=2, n_redundant=0,
                           n_clusters_per_class=1, random_state=42)

# 只保留15个标签
n_labeled = 15
rng = np.random.RandomState(42)
mask_unlabeled = rng.rand(len(y)) > (n_labeled / len(y))
y_semi = y.copy()
y_semi[mask_unlabeled] = -1

# ---- 2. 监督SVM基线：只用15个有标签样本 ----
X_labeled = X[~mask_unlabeled]
y_labeled = y[~mask_unlabeled]

svm = SVC(kernel='linear', C=1.0, random_state=42)
svm.fit(X_labeled, y_labeled)
y_pred_svm = svm.predict(X)
acc_svm = accuracy_score(y, y_pred_svm)
print(f"纯监督SVM (仅{n_labeled}个标签): {acc_svm:.4f}")
print(f"  ⚠️ 只用15个样本来画分界线 → 分界线位置完全不可靠")
```

```python
# ============================================
# 📌 模拟 TSVM 的"伪标签迭代"机制 — 从零实现
# ============================================
# sklearn 没有内置 TSVM，这里实现其核心逻辑：
# 1. 用有标签数据训练初始SVM
# 2. 用初始SVM给无标签数据打伪标签
# 3. 交换"最可疑"的样本标签，重新训练
# 4. 重复直到分界面稳定

class ToyTSVM:
    """
    模拟TSVM核心机制的简易实现
    实现了"局部搜索"策略：通过交换高松弛变量的样本来优化分界面

    比喻：
    你先用少数学霸的答案画了一条分界线
    然后看其余同学落在线的哪一边，给他们"暂定"成绩
    接着找那些"落在线旁边很尴尬"的同学（松弛变量大的）
    试试交换他们的成绩假设，看看能不能画出一条更宽的"无争议地带"
    """

    def __init__(self, C_l=10.0, C_u=0.01, max_iter=20):
        """
        C_l: 有标签样本的惩罚系数（很大 → "零容忍"）
        C_u: 无标签样本的惩罚系数（很小 → "极度宽容"）
        🌟 这就是为什么 C_u << C_l：
        你有绝对把握的是有标签数据，它们必须待在隔离带外面
        无标签数据的伪标签只是"猜测"，允许它们犯错
        """
        self.C_l = C_l      # 有标签样本的"高贵系数"
        self.C_u = C_u      # 无标签样本的"草根系数"
        self.max_iter = max_iter

    def fit(self, X, y_semi):
        """
        X: 全部数据
        y_semi: 半监督标签，-1=未标记
        """
        X = np.array(X)
        y_semi = np.array(y_semi).astype(float)

        labeled_mask = y_semi != -1
        unlabeled_mask = y_semi == -1

        X_l, y_l = X[labeled_mask], y_semi[labeled_mask]
        X_u = X[unlabeled_mask]
        n_u = len(X_u)

        # ---- Step 1: 用有标签数据训练初始SVM ----
        svm = SVC(kernel='linear', C=1.0, random_state=42)
        svm.fit(X_l, y_l)

        # ---- Step 2: 给无标签数据打伪标签 ----
        y_u_pred = svm.predict(X_u)

        # ---- Step 3: 迭代优化 — 寻找并交换可疑样本 ----
        for it in range(self.max_iter):
            # 合并有标签和伪标签数据
            X_all = np.vstack([X_l, X_u])
            y_all = np.hstack([y_l, y_u_pred])

            # 构建样本权重（反映 C_l vs C_u 的区别）
            sample_weight = np.ones(len(X_all))
            sample_weight[:len(X_l)] = self.C_l      # 有标签 → 权重 = C_l (大)
            sample_weight[len(X_l):] = self.C_u      # 无标签 → 权重 = C_u (小)
            # 💡 这就是 C_l >> C_u 在代码中的体现！
            # 有标签样本的"犯错代价"远大于无标签样本

            svm = SVC(kernel='linear', C=1.0, random_state=42)
            svm.fit(X_all, y_all, sample_weight=sample_weight)

            # ---- 找"最可疑"的无标签样本 ----
            # 可疑标准：松弛变量 ξ_i + ξ_j > 2 的两个异类样本
            # 通俗理解：两个被预测为不同类但都"跨过线"的样本
            decision = svm.decision_function(X_u)  # 到决策边界的带符号距离

            # 计算每个样本的 ξ — 负号表示在错误一侧
            xi = np.maximum(0, 1 - y_u_pred * decision)

            # 找松弛最大的样本对（来自不同预测类别）
            # ⚠️ 这里的逻辑是：找最"站在线旁边很尴尬"的那些样本
            if len(xi) > 1:
                worst_idx = np.argsort(xi)[-2:]  # 松弛最大的两个
                if y_u_pred[worst_idx[0]] != y_u_pred[worst_idx[1]]:
                    # 它们的松弛之和超过了阈值 → 高度可疑！
                    xi_sum = xi[worst_idx[0]] + xi[worst_idx[1]]
                    if xi_sum > 2:
                        # 🌟 交换它们的标签！试试图会不会更好
                        y_u_pred[worst_idx] = -y_u_pred[worst_idx]

        self.svm_ = svm
        self.X_l_ = X_l
        self.y_l_ = y_l
        self.X_u_ = X_u
        self.y_u_pred_ = y_u_pred

    def predict(self, X):
        return self.svm_.predict(X)

    def decision_function(self, X):
        return self.svm_.decision_function(X)


# ---- 测试模拟TSVM ----
ttsvm = ToyTSVM(C_l=10.0, C_u=0.01, max_iter=30)
ttsvm.fit(X, y_semi)
y_pred_tsvm = ttsvm.predict(X)
acc_tsvm = accuracy_score(y, y_pred_tsvm)

print(f"\n模拟TSVM (仅{n_labeled}个标签 + 285个无标签): {acc_tsvm:.4f}")
print(f"纯监督SVM (仅{n_labeled}个标签):               {acc_svm:.4f}")
print(f"💡 TSVM利用无标签数据后，分界面更准确地穿过了低密度区域")
```

# 🌟 核心对比：C_u << C_l 的效果实验
# ============================================

def tsVM_Cu_comparison(X, y_semi, y_true, C_u_values=[0.001, 0.01, 0.1, 1.0, 10.0]):
    """对比不同 C_u 设置下 TSVM 的表现"""
    print("\n📌 C_u 取值对TSVM的影响 (C_l 固定为 10):")
    print("-" * 60)
    for Cu in C_u_values:
        ttsvm = ToyTSVM(C_l=10.0, C_u=Cu, max_iter=30)
        ttsvm.fit(X, y_semi)
        acc = accuracy_score(y_true, ttsvm.predict(X))
        label = ""
        if Cu < 1.0:
            label = "  ← 🌟 推荐：无标签数据说话小声，有标签数据一语定乾坤"
        elif Cu == 1.0:
            label = "  ← ⚠️ 有标签和无标签平起平坐，等于放弃了先验"
        else:
            label = "  ← 🚨 无标签伪标签比真实标签还大声，分界线会被带偏"
        print(f"  C_u = {Cu:6.3f} → 准确率 = {acc:.4f}{label}")


tsVM_Cu_comparison(X, y_semi, y)

```python
# 📌 TSVM优化目标的核心矛盾与解法：
# 
# 理论上要穷举所有 2^|未标记样本| 种标签组合（天文数字！）
# 实际上 TSVM 用"局部搜索"策略来逼近最优解：
# 
# 第1步：用有标签数据训练初始分类器
# 第2步：给无标签样本打伪标签
# 第3步：找那些"站在线旁边很不舒服"的样本（ξ_i 大的）
# 第4步：试试给它们换标签，看能不能让分界线更宽
# 第5步：重复 3-4 直到收敛
# 
# 🌟 比喻：就像排座位
# 你先确定少数几个同学的座位（有标签），然后让大家自己找位置坐下
# 接着看谁坐得太挤了（ξ 大），试着让两个"挤在一起但可能不是一队"的人
# 换位子，看看整排过道是不是宽了

# ============================================
# 类别不平衡的处理
# ============================================

# 📌 公式 u_+ * C_u^+ = u_- * C_u^- 的含义：
# 给"人少的类别"更大的惩罚系数，保证分界线不会被"人多的类别"带偏
# 
# 通俗比喻：
# 班里男生30个，女生3个
# 如果每人一票选举班干部，女生注定被忽略
# 所以给女生每人10票（增大C_u^-），让双方投票总量相等
# 这样分界线（决策）才能公平

# 代码实现思路：
def balance_tsvm_penalty(y_pseudo):
    """
    根据伪标签分布，给少数类更大的惩罚系数
    """
    n_pos = np.sum(y_pseudo == 1)
    n_neg = np.sum(y_pseudo == -1)

    # 🌟 核心：人少的类别获得更大权重
    C_u_pos = n_neg / (n_pos + 1e-8)   # 正类少则正类权重高
    C_u_neg = n_pos / (n_neg + 1e-8)   # 负类少则负类权重高

    # 对每个样本构建权重
    sample_C_u = np.where(y_pseudo == 1, C_u_pos, C_u_neg)
    print(f"正类样本数={n_pos}, 负类样本数={n_neg}")
    print(f"  正类惩罚系数 C_u^+ = {C_u_pos:.2f}")
    print(f"  负类惩罚系数 C_u^- = {C_u_neg:.2f}")
    print(f"  💡 验证：{n_pos} * {C_u_pos:.2f} ≈ {n_neg} * {C_u_neg:.2f} (平衡)")

    return sample_C_u


y_demo = np.array([1, 1, 1, -1, -1, -1, -1, -1])  # 3正5负，不平衡
sample_C_u = balance_tsvm_penalty(y_demo)
```

## 三、松弛变量与可疑样本标记

> 松弛变量的物理含义是什么？它如何帮助TSVM迭代优化？

```python
# 📌 松弛变量 ξ 的物理含义：
# ξ=0  → 乖乖待在隔离带自己那一侧（好孩子）
# ξ>0  → 跨进了隔离带甚至跑到了对面（违规了！）
# ξ越大 → "违规"越严重
#
# S3VM 的关键洞察：ξ_i + ξ_j > 2 时，这两个点很可能是被分错的
# 因为它们在分界线的"错误一侧"或者刚刚擦边站在隔离带里
# 
# 代码检测逻辑：
decision_values = ttsvm.decision_function(ttsvm.X_u_)
y_pred_u = ttsvm.y_u_pred_
xi_values = np.maximum(0, 1 - y_pred_u * decision_values)  # hinge损失

print(f"\n📌 无标签样本的松弛变量统计:")
print(f"  最小ξ = {xi_values.min():.4f}")
print(f"  最大ξ = {xi_values.max():.4f}")
print(f"  平均ξ = {xi_values.mean():.4f}")
print(f"  ξ>1的样本数 = {np.sum(xi_values > 1)} (最可疑的)")
# 💡 这些ξ>1的样本就是TSVM迭代中最可能被交换标签的对象
```
