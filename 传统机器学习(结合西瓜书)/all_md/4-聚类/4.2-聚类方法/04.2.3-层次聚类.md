## 一、AGNES 层次聚类 — sklearn 实战

> 如何通过自底向上的聚合策略，逐步将样本合并成层次化的簇结构？

![alt text](image-12.png)

```python
import numpy as np
from sklearn.cluster import AgglomerativeClustering
from sklearn.datasets import make_blobs
from scipy.cluster.hierarchy import dendrogram, linkage, fcluster
from sklearn.metrics import silhouette_score

# ============================================
# 🌟 AGNES 核心思想：每个人都是一座孤岛，慢慢合并成大陆
# ============================================
# 第 1 步：每个样本自己就是一个簇（m 个样本 = m 个簇）
# 第 2 步：算所有簇两两之间的距离 → 构造距离矩阵 M
# 第 3 步：找距离最近的两个簇 → 合并成一个新簇
# 第 4 步：更新距离矩阵（重新算新簇和其他簇的距离）
# 第 5 步：重复 3-4，直到只剩下 k 个簇
# 💡 整个过程就像"城市合并"——从村到镇到县到市，逐级聚合

# ============================================
# 🚀 sklearn 一行搞定
# ============================================
X, _ = make_blobs(n_samples=200, n_features=2,
                  centers=4, cluster_std=0.8, random_state=42)

agg = AgglomerativeClustering(n_clusters=4, linkage='ward')
labels = agg.fit_predict(X)

print(f"AGNES 聚类结果: {np.bincount(labels)}")
print(f"轮廓系数: {silhouette_score(X, labels):.4f}")

# ============================================
# 📌 三种链接方式 (linkage) — 决定"两个簇之间距离"怎么算
# ============================================
# 这就像衡量两个部落之间"距离"的不同标准：
#
# 单链接 (single)：    取两部落边界最近的两间房 → min
# 全链接 (complete)：  取两部落最远的两间房 → max
# 均链接 (average)：   取两部落所有人两两见面的平均距离 → avg
# Ward (ward)：        合并后使方差增量最小的两个簇（最常用！）

print("\n=== 不同链接方式对比 ===")
for linkage_type in ['single', 'complete', 'average', 'ward']:
    a = AgglomerativeClustering(n_clusters=4, linkage=linkage_type)
    labs = a.fit_predict(X)
    try:
        sil = silhouette_score(X, labs)
    except:
        sil = -1
    print(f"linkage='{linkage_type:7s}': 轮廓系数={sil:.4f}")

# 💡 Ward 通常表现最好，因为它追求簇内方差最小化，产生紧凑均匀的球形簇
```

## 二、树状图 (Dendrogram) — 层次聚类的可视化

> 层次聚类的合并过程如何可视化？如何从树状图中选择最佳簇数？

```python
# ============================================
# 🌟 树状图是层次聚类最漂亮的输出
#    横轴 = 样本，纵轴 = 合并时的距离
#    低处合并的是"近亲"，高处合并的是"远亲"
#    "一刀切"下去 → 切出多少个簇
# ============================================

from scipy.cluster.hierarchy import dendrogram, linkage
import matplotlib.pyplot as plt

# scipy 的 linkage 函数返回层次聚类所需的合并信息
# method 参数对应 linkage 方式
Z = linkage(X, method='ward')  # Z 是一个 (m-1)×4 的矩阵

# 查看树状图的前几条合并记录
print("前 5 次合并 (cluster_i, cluster_j, 距离, 合并后簇大小):")
print(Z[:5])
# 每行含义：[簇A编号, 簇B编号, 合并距离, 合并后总样本数]

#    找纵轴上"跳变最大"的地方——距离突然大幅增加说明在强行合并不相似的簇
#    在那个跳变点下面"切一刀"得到的就是最自然的簇数
```

## 三、AGNES 算法细节 — numpy 手动模拟

> AGNES 的内部循环是如何工作的？"贪心合并 + 矩阵维护"的机制是什么？

```python
# ============================================
# 📌 手动实现 AGNES 的核心循环
#    感受"贪心合并 + 矩阵维护"的机制
# ============================================

class SimpleAGNES:
    """
    手动 AGNES，展示自底向上层次聚类的内部运作

    🌟 核心流程：
    1. 每个点各自为簇
    2. 计算距离矩阵 M（簇间两两距离）
    3. 循环：找 M 中最小的值 → 合并对应两个簇 → 更新 M → 减少簇数
    """
    def __init__(self, n_clusters=3, linkage='single'):
        self.n_clusters = n_clusters
        self.linkage = linkage

    def _cluster_dist(self, cluster_i, cluster_j, points, clusters):
        """计算两个簇之间的距离（根据 linkage 策略）"""
        pts_i = points[clusters == cluster_i]
        pts_j = points[clusters == cluster_j]

        # 所有点对之间的欧氏距离
        dists = np.sqrt(np.sum((pts_i[:, np.newaxis] - pts_j[np.newaxis, :]) ** 2, axis=-1))

        if self.linkage == 'single':
            return dists.min()    # 两簇最近的两个点
        elif self.linkage == 'complete':
            return dists.max()    # 两簇最远的两个点
        elif self.linkage == 'average':
            return dists.mean()   # 所有点对的平均距离
        else:
            raise ValueError(f"Unknown linkage: {self.linkage}")

    def fit_predict(self, X):
        X = np.array(X)
        n = len(X)

        # 🌟 第 1 步：每个样本自己是一个簇
        labels = np.arange(n)  # 标签从 0 到 n-1，每个样本独占一簇
        n_clusters = n

        # 迭代合并，直到剩余 n_clusters 个簇
        while n_clusters > self.n_clusters:
            # 🌟 找出距离最近的两个簇
            unique_labels = np.unique(labels)
            min_dist = float('inf')
            merge_pair = (None, None)

            for i in range(len(unique_labels)):
                for j in range(i + 1, len(unique_labels)):
                    d = self._cluster_dist(unique_labels[i], unique_labels[j], X, labels)
                    if d < min_dist:
                        min_dist = d
                        merge_pair = (unique_labels[i], unique_labels[j])

            # 🌟 合并：把簇 j 的标签全改成簇 i 的标签
            labels[labels == merge_pair[1]] = merge_pair[0]
            n_clusters -= 1

        # 重新映射标签为 0, 1, 2, ...
        _, labels = np.unique(labels, return_inverse=True)
        return labels

# 测试手动实现
simple_agnes = SimpleAGNES(n_clusters=4, linkage='single')
manual_labels = simple_agnes.fit_predict(X)
print(f"手动 AGNES: {np.bincount(manual_labels)}")
print(f"轮廓系数: {silhouette_score(X, manual_labels):.4f}")
```

## 四、三种链接方式的"性格"对比

> 不同链接方式对聚类结果有什么影响？"桥梁效应"是什么？

```python
# ============================================
# 📌 链接方式决定了层次聚类的"性格"
# ============================================
# 视觉化三个族的合并逻辑示例

# 模拟两个拉长的簇 + 一个噪声点
np.random.seed(42)
cluster_a = np.random.randn(30, 2) * 0.3 + [-3, 0]       # 左边一团
cluster_b = np.random.randn(30, 2) * 0.3 + [3, 0]        # 右边一团
noise = np.array([[0, 0]])                                 # 中间一个噪声点（"桥梁"）
X_bridge = np.vstack([cluster_a, cluster_b, noise])

print("=== 噪声'桥梁'实验 ===")
# ⚠️  single-linkage 最怕"桥梁效应"——中间一个噪声点就把两个远簇连接了
for method in ['single', 'complete', 'average', 'ward']:
    Z_method = linkage(X_bridge, method=method)
    # 在 n_clusters=2 时切一刀
    labs_method = fcluster(Z_method, t=2, criterion='maxclust')

    # 检查噪声点是否单独成簇（正确）还是被合并进了某个簇（错误）
    noise_label = labs_method[-1]  # 最后一个点是噪声
    unique, counts = np.unique(labs_method, return_counts=True)
    # 噪声应该单独成簇（count=1）或被合理地归入某个簇
    print(f"{method:8s}: 簇分布={dict(zip(unique, counts))}, 噪声点在簇 {noise_label}")

# 💡 结论：
#   single:  链式效应，能发现细长簇，但怕噪声桥接
#   complete: 倾向产生紧凑均匀的球形簇，抗噪声效果好
#   average:  最稳健，综合考虑所有点，科研中最常用
#   ward:     最小方差增量，效果通常最好，适合球形数据

# ============================================
# ⚠️  层次聚类的局限：
#   1. 计算复杂度 O(n^3)（或优化后 O(n^2 log n)），大数据集吃力
#   2. 一旦合并就不能撤销——一步错步步错（贪心策略的代价）
#   3. 不同 linkage 可能得出完全不同的树状图
#   4. 不直接处理噪声点（不像 DBSCAN 有 -1 标签）
# ============================================
```

## 五、层次聚类 vs 原型聚类 vs 密度聚类 — 三剑客对比

> 三种聚类方法在同一数据上的表现如何？各自适合什么场景？

```python
from sklearn.cluster import KMeans, DBSCAN, AgglomerativeClustering
from sklearn.metrics import adjusted_rand_score

# ============================================
# 🌟 同一份数据，三种聚类方法的对比
# ============================================
X_all, y_all = make_blobs(n_samples=400, n_features=2,
                           centers=4, cluster_std=[0.4, 0.8, 1.2, 0.6],
                           random_state=42)

# 三种方法各跑一遍
results = {
    'KMeans':    KMeans(n_clusters=4, random_state=42, n_init='auto').fit_predict(X_all),
    'AGNES':     AgglomerativeClustering(n_clusters=4, linkage='ward').fit_predict(X_all),
    'DBSCAN':    DBSCAN(eps=1.5, min_samples=5).fit_predict(X_all),
}

# 两两比较一致性 (ARI)
print("=== 三种聚类方法的 ARI 对比 ===")
for name1, labels1 in results.items():
    for name2, labels2 in results.items():
        if name1 < name2:
            # 排除 DBSCAN 的噪声点（标签=-1）
            if -1 in labels2:
                valid = labels2 != -1
                ari = adjusted_rand_score(labels1[valid], labels2[valid])
            else:
                ari = adjusted_rand_score(labels1, labels2)
            print(f"{name1:7s} vs {name2:7s}: ARI = {ari:.4f}")

# 📌 总结三类方法的适用场景：
# K-means  → 球形簇，已知 k，速度快 → 第一选择
# GMM      → 重叠数据，需要概率归属（软聚类）
# DBSCAN   → 任意形状，含噪声，不知道 k → 探索性分析首选
# AGNES    → 想看到"层级结构"，数据量适中 → 需要树状图时用
```
