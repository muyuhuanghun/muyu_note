# 六.Server API 与应用集成

📌 llama-server 是 llama.cpp 的核心组件——提供 REST API 接口，让你的本地模型可以被任何应用调用。

# 一.启动 Server

## 1.基础启动命令

```bash
# 📌 启动 llama-server

# 基础启动
./llama-server -m model.gguf -ngl 99 -c 8192

# 自定义端口和主机
./llama-server -m model.gguf \
    -ngl 99 \
    -c 8192 \
    --port 8080 \
    --host 0.0.0.0

# 🌟 参数说明：
# --port 8080     → 监听端口
# --host 0.0.0.0  → 允许局域网访问
# --host 127.0.0.1 → 仅本机访问（默认）

# 💡 启动后访问：
# 本机：http://localhost:8080
# 局域网：http://<你的IP>:8080
```

## 2.高级启动配置

```bash
# 📌 生产环境推荐配置

./llama-server -m model.gguf \
    -ngl 99 \
    -c 8192 \
    --port 8080 \
    --host 0.0.0.0 \
    --flash-attn on \
    --ctx-size 8192 \
    --parallel 4 \
    --cont-batching

# 🌟 关键参数：
# --parallel 4       → 支持 4 个并发请求
# --cont-batching    → 连续批处理（提升吞吐量）
# --flash-attn on    → 降低显存占用

# 💡 并发处理：
# --parallel 1 → 单用户（默认）
# --parallel 2 → 双用户
# --parallel 4 → 四用户（需要更多显存）
```

# 三.Python 集成

## 1.使用 requests 库

```python
# 📌 Python 调用 llama-server

import requests

def chat_with_llama(prompt, system_prompt="你是一个有帮助的助手"):
    """调用 llama-server 的 Chat API"""
    url = "http://localhost:8080/v1/chat/completions"

    payload = {
        "model": "qwen2",
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": prompt}
        ],
        "temperature": 0.7,
        "max_tokens": 512
    }

    response = requests.post(url, json=payload)
    return response.json()["choices"][0]["message"]["content"]

# 使用示例
answer = chat_with_llama("解释一下什么是量子计算")
print(answer)
```

## 2.流式输出实现

```python
# 📌 Python 流式输出

import requests

def stream_chat(prompt):
    """流式调用 llama-server"""
    url = "http://localhost:8080/v1/chat/completions"

    payload = {
        "model": "qwen2",
        "messages": [
            {"role": "user", "content": prompt}
        ],
        "stream": True
    }

    response = requests.post(url, json=payload, stream=True)

    for line in response.iter_lines():
        if line:
            line = line.decode('utf-8')
            if line.startswith('data: '):
                data = line[6:]
                if data == '[DONE]':
                    break
                # 解析并输出内容
                import json
                chunk = json.loads(data)
                content = chunk["choices"][0]["delta"].get("content", "")
                print(content, end="", flush=True)

# 使用示例
stream_chat("写一个快速排序算法")
```

## 2.流式输出实现

```javascript
// 📌 JavaScript 流式输出

async function streamChat(prompt) {
    const response = await fetch('http://localhost:8080/v1/chat/completions', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({
            model: 'qwen2',
            messages: [
                { role: 'user', content: prompt }
            ],
            stream: true
        })
    });

    const reader = response.body.getReader();
    const decoder = new TextDecoder();

    while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        const chunk = decoder.decode(value);
        const lines = chunk.split('\n');

        for (const line of lines) {
            if (line.startsWith('data: ')) {
                const data = line.slice(6);
                if (data === '[DONE]') break;

                const json = JSON.parse(data);
                const content = json.choices[0].delta?.content || '';
                process.stdout.write(content);
            }
        }
    }
}

// 使用示例
streamChat('解释递归的概念');
```

# 五.应用集成示例

## 1.命令行聊天机器人

```python
# 📌 简单的命令行聊天机器人

import requests

def chat_bot():
    """交互式聊天机器人"""
    print("🤖 本地 AI 助手（输入 'quit' 退出）")
    print("-" * 40)

    history = []

    while True:
        user_input = input("\n👤 你: ")
        if user_input.lower() == 'quit':
            break

        history.append({"role": "user", "content": user_input})

        response = requests.post(
            "http://localhost:8080/v1/chat/completions",
            json={
                "model": "qwen2",
                "messages": history,
                "temperature": 0.7,
                "max_tokens": 512
            }
        )

        assistant_message = response.json()["choices"][0]["message"]["content"]
        history.append({"role": "assistant", "content": assistant_message})

        print(f"\n🤖 AI: {assistant_message}")

if __name__ == "__main__":
    chat_bot()
```

# 六.常见问题排查

```bash
# 📌 Server 常见问题

# 问题 1：端口被占用
# 解决：换一个端口
./llama-server -m model.gguf --port 8081

# 问题 2：局域网无法访问
# 检查：
# 1. 是否使用 --host 0.0.0.0
# 2. 防火墙是否放行端口
# 3. 网络是否连通

# 问题 3：响应速度慢
# 优化：
# 1. 增加 --parallel 值
# 2. 启用 --cont-batching
# 3. 减小 -c 上下文长度

# 问题 4：显存不足
# 解决：
# 1. 减小 -ngl 值
# 2. 使用 --flash-attn
# 3. 降低 --parallel 值
```
