# 一.RAG（检索增强生成）

## 1.什么是 RAG

```bash
# 📌 RAG 核心概念

# 传统 LLM：
# 用户提问 → 模型凭记忆回答（可能幻觉）

# RAG 模式：
# 用户提问 → 检索相关文档 → 将文档+问题一起给模型 → 准确回答

# 🌟 优势：
# - 减少幻觉
# - 回答有据可依
# - 可以处理私有数据
# - 知识可实时更新
```

## 2.RAG 系统架构

```python
# 📌 RAG 系统核心组件

# 1. 文档加载器（Document Loader）
# 2. 文本分割器（Text Splitter）
# 3. 向量数据库（Vector Store）
# 4. 检索器（Retriever）
# 5. LLM（llama.cpp）

# 🌟 流程：
# 文档 → 分割 → 向量化 → 存储 → 检索 → 生成
```

## 3.使用 LangChain 构建 RAG

```python
# 📌 LangChain + llama.cpp RAG 实现

from langchain_community.llms import LlamaCpp
from langchain_community.document_loaders import TextLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain.chains import RetrievalQA

# 1. 加载文档
loader = TextLoader("my_documents.txt", encoding="utf-8")
documents = loader.load()

# 2. 分割文本
text_splitter = RecursiveCharacterTextSplitter(
    chunk_size=500,
    chunk_overlap=50
)
docs = text_splitter.split_documents(documents)

# 3. 创建向量数据库
embeddings = HuggingFaceEmbeddings(model_name="all-MiniLM-L6-v2")
vectorstore = FAISS.from_documents(docs, embeddings)

# 4. 加载 LLM
llm = LlamaCpp(
    model_path="models/qwen2.5-7b-q4_k_m.gguf",
    n_gpu_layers=99,
    n_ctx=8192,
    temperature=0.3,
    verbose=False
)

# 5. 创建 RAG 链
qa_chain = RetrievalQA.from_chain_type(
    llm=llm,
    chain_type="stuff",
    retriever=vectorstore.as_retriever(search_kwargs={"k": 3})
)

# 6. 提问
answer = qa_chain.invoke("什么是机器学习？")
print(answer["result"])

# 💡 关键参数：
# chunk_size → 文本块大小（太大太小都不好）
# k → 检索的文档数量（通常 3-5）
# temperature → 生成温度（RAG 建议用低温度）
```

## 4.使用 llama.cpp Embeddings API

```python
# 📌 使用 llama-server 的 Embeddings API

import requests
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

def get_embedding(text):
    """获取文本的向量表示"""
    response = requests.post(
        "http://localhost:8080/v1/embeddings",
        json={"model": "qwen2", "input": text}
    )
    return response.json()["data"][0]["embedding"]

def search_documents(query, documents, top_k=3):
    """检索最相关的文档"""
    query_embedding = get_embedding(query)

    similarities = []
    for doc in documents:
        doc_embedding = get_embedding(doc)
        similarity = cosine_similarity(
            [query_embedding],
            [doc_embedding]
        )[0][0]
        similarities.append((doc, similarity))

    # 按相似度排序
    similarities.sort(key=lambda x: x[1], reverse=True)
    return similarities[:top_k]

# 使用示例
documents = [
    "机器学习是人工智能的一个分支",
    "深度学习使用神经网络",
    "Python 是流行的编程语言"
]

query = "什么是 AI？"
results = search_documents(query, documents)

for doc, score in results:
    print(f"相似度: {score:.3f} | 内容: {doc}")
```

# 二.函数调用（Function Calling）

## 1.什么是函数调用

```bash
# 📌 函数调用概念

# 传统模式：
# 用户："今天天气怎么样？"
# AI："我无法获取实时天气信息"

# 函数调用模式：
# 用户："今天天气怎么样？"
# AI：调用 get_weather() 函数
# AI："今天北京晴，25°C"

# 🌟 优势：
# - 让 AI 能够执行实际操作
# - 获取实时信息
# - 与外部系统集成
```

## 2.函数定义格式

```python
# 📌 函数定义（OpenAI 格式）

functions = [
    {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "获取指定城市的天气信息",
            "parameters": {
                "type": "object",
                "properties": {
                    "city": {
                        "type": "string",
                        "description": "城市名称，如：北京、上海"
                    }
                },
                "required": ["city"]
            }
        }
    },
    {
        "type": "function",
        "function": {
            "name": "search_web",

            "parameters": {
                "type": "object",
                "properties": {
                    "query": {
                        "type": "string",

                },
                "required": ["query"]

        }
    }
]
```

## 3.实现函数调用

```python
# 📌 函数调用完整实现

import requests
import json

def get_weather(city):
    """模拟获取天气"""
    # 实际应用中调用天气 API
    weather_data = {
        "北京": {"temp": 25, "condition": "晴"},
        "上海": {"temp": 28, "condition": "多云"},
    }
    return weather_data.get(city, {"temp": 0, "condition": "未知"})

def chat_with_functions(user_message):
    """带函数调用的对话"""
    response = requests.post(
        "http://localhost:8080/v1/chat/completions",
        json={
            "model": "qwen2",
            "messages": [
                {"role": "user", "content": user_message}
            ],
            "functions": functions,
            "function_call": "auto"
        }
    )

    result = response.json()
    message = result["choices"][0]["message"]

    # 检查是否需要调用函数
    if message.get("function_call"):
        func_name = message["function_call"]["name"]
        func_args = json.loads(message["function_call"]["arguments"])

        # 执行函数
        if func_name == "get_weather":
            func_result = get_weather(func_args["city"])
        else:
            func_result = {"error": "未知函数"}

        # 将函数结果返回给模型
        response = requests.post(
            "http://localhost:8080/v1/chat/completions",
            json={
                "model": "qwen2",
                "messages": [
                    {"role": "user", "content": user_message},
                    message,
                    {
                        "role": "function",
                        "name": func_name,
                        "content": json.dumps(func_result, ensure_ascii=False)
                    }
                ]
            }
        )
        return response.json()["choices"][0]["message"]["content"]

    return message["content"]

# 使用示例
answer = chat_with_functions("今天北京天气怎么样？")
print(answer)
```

import whisper
import subprocess

def transcribe_audio(audio_path):
    """将音频转换为文本"""
    model = whisper.load_model("base")
    result = model.transcribe(audio_path, language="zh")

## 4.完整语音对话系统

```python
# 📌 语音对话完整实现

import whisper
import edge_tts
import asyncio
import requests
import os

class VoiceAssistant:
    def __init__(self, server_url="http://localhost:8080"):
        self.server_url = server_url
        self.whisper_model = whisper.load_model("base")
        self.conversation_history = []

    def listen(self, audio_path):
        """语音识别"""
        result = self.whisper_model.transcribe(audio_path, language="zh")
        return result["text"]

    def think(self, user_input):
        """AI 思考"""
        self.conversation_history.append({
            "role": "user",
            "content": user_input
        })

        response = requests.post(
            f"{self.server_url}/v1/chat/completions",
            json={
                "model": "qwen2",
                "messages": self.conversation_history,
                "temperature": 0.7,
                "max_tokens": 512
            }
        )

        assistant_message = response.json()["choices"][0]["message"]["content"]
        self.conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })

        return assistant_message

    async def speak(self, text, output_file="response.mp3"):
        """语音合成"""
        communicate = edge_tts.Communicate(text, "zh-CN-XiaoxiaoNeural")
        await communicate.save(output_file)
        # 播放音频（需要额外库）
        os.system(f"start {output_file}")  # Windows

    async def chat(self, audio_path):
        """完整对话流程"""
        # 1. 语音识别
        user_text = self.listen(audio_path)
        print(f"👤 用户: {user_text}")

        # 2. AI 思考
        response = self.think(user_text)
        print(f"🤖 AI: {response}")

        # 3. 语音合成
        await self.speak(response)

# 使用示例
async def main():
    assistant = VoiceAssistant()
    await assistant.chat("user_recording.mp3")

asyncio.run(main())
```

import requests
import base64

def analyze_image(image_path, question="描述这张图片"):
    """分析图片内容"""
    # 将图片转为 base64
    with open(image_path, "rb") as f:
        image_data = base64.b64encode(f.read()).decode("utf-8")

    response = requests.post(
        "http://localhost:8080/v1/chat/completions",
        json={
            "model": "qwen2-vl",
            "messages": [
                {
                    "role": "user",
                    "content": [
                        {"type": "text", "text": question},
                        {
                            "type": "image_url",
                            "image_url": {
                                "url": f"data:image/jpeg;base64,{image_data}"
                            }
                        }
                    ]
                }
            ],
            "max_tokens": 512
        }
    )

    return response.json()["choices"][0]["message"]["content"]

# 使用示例

```python
# 📌 ChromaDB 知识库实现

import chromadb
from chromadb.utils import embedding_functions
import requests

class KnowledgeBase:
    def __init__(self, server_url="http://localhost:8080"):
        self.server_url = server_url
        self.client = chromadb.PersistentClient(path="./knowledge_db")
        self.collection = self.client.get_or_create_collection(
            name="my_knowledge",
            embedding_function=embedding_functions.DefaultEmbeddingFunction()
        )

    def add_documents(self, documents, metadatas=None):
        """添加文档到知识库"""
        ids = [f"doc_{i}" for i in range(len(documents))]
        self.collection.add(
            documents=documents,
            metadatas=metadatas or [{}] * len(documents),
            ids=ids
        )
        print(f"✅ 已添加 {len(documents)} 条文档")

    def search(self, query, n_results=3):
        """检索相关文档"""
        results = self.collection.query(
            query_texts=[query],
            n_results=n_results
        )
        return results["documents"][0]

    def ask(self, question):
        """基于知识库问答"""
        # 1. 检索相关文档
        relevant_docs = self.search(question)
        context = "\n".join(relevant_docs)

        # 2. 构建 prompt
        prompt = f"""基于以下信息回答问题：

{context}

问题：{question}
回答："""

        # 3. 调用 LLM
        response = requests.post(
            f"{self.server_url}/v1/chat/completions",
            json={
                "model": "qwen2",
                "messages": [
                    {"role": "user", "content": prompt}
                ],
                "temperature": 0.3,
                "max_tokens": 512
            }
        )

        return response.json()["choices"][0]["message"]["content"]

# 使用示例
kb = KnowledgeBase()

# 添加文档
kb.add_documents([
    "机器学习是人工智能的一个分支",
    "深度学习使用多层神经网络",
    "Python 是流行的 AI 编程语言"
])

# 提问
answer = kb.ask("什么是机器学习？")
print(answer)
```
