# 二.快速切换模型

## 1.脚本化切换

```bash
#!/bin/bash
# 📌 模型切换脚本

# 模型配置
declare -A MODELS
MODELS[qwen2.5-7b]="models/qwen2.5-7b/qwen2.5-7b-q4_k_m.gguf"
MODELS[llama3-8b]="models/llama3-8b/llama3-8b-q4_k_m.gguf"
MODELS[gemma2-9b]="models/gemma2-9b/gemma2-9b-q4_k_m.gguf"

# 显示可用模型
show_models() {
    echo "📌 可用模型："
    for name in "${!MODELS[@]}"; do
        echo "  - $name"
    done
}

# 启动指定模型
start_model() {
    local model_name=$1
    local model_path=${MODELS[$model_name]}

    if [ -z "$model_path" ]; then
        echo "❌ 模型不存在: $model_name"
        show_models
        return 1
    fi

    echo "🚀 启动模型: $model_name"
    ./llama-server -m "$model_path" -ngl 99 -c 8192 --port 8080
}

# 主函数
case "$1" in
    list)
        show_models
        ;;
    start)
        start_model "$2"
        ;;
    *)
        echo "用法: $0 list|start <模型名>"
        ;;
esac
```

```python
# 📌 Python 读取配置并启动模型

import json
import subprocess

def load_config(config_path="models.json"):
    """加载模型配置"""
    with open(config_path, 'r', encoding='utf-8') as f:
        return json.load(f)

def start_model(model_name=None):
    """启动指定模型"""
    config = load_config()

    if model_name is None:
        model_name = config.get("default", list(config["models"].keys())[0])

    if model_name not in config["models"]:
        print(f"❌ 模型不存在: {model_name}")
        print(f"📌 可用模型: {', '.join(config['models'].keys())}")
        return

    model = config["models"][model_name]
    print(f"🚀 启动模型: {model_name}")
    print(f"📝 {model['description']}")

    cmd = [
        "./llama-server",
        "-m", model["path"],
        "-ngl", str(model["ngl"]),
        "-c", str(model["context"]),
        "--port", "8080"
    ]

    subprocess.run(cmd)

# 使用示例
if __name__ == "__main__":
    import sys
    model_name = sys.argv[1] if len(sys.argv) > 1 else None
    start_model(model_name)
```

# 四.常见问题排查

## 1.显存不足（OOM）

```bash
# 📌 显存不足问题排查

# 症状：
# - CUDA out of memory
# - 模型加载失败
# - 推理时崩溃

# 解决方案：
# 1. 减少 GPU 层数
./llama-cli -m model.gguf -ngl 20  # 从 99 降到 20

# 2. 使用 Flash Attention
./llama-cli -m model.gguf -ngl 99 --flash-attn on

# 3. 减小上下文长度
./llama-cli -m model.gguf -ngl 99 -c 4096  # 从 8192 降到 4096

# 4. 使用量化 KV Cache
./llama-cli -m model.gguf -ngl 99 \
    --cache-type-k q8_0 --cache-type-v q8_0

# 5. 换用更小的量化版本
# Q5_K_M → Q4_K_M → Q3_K_M → IQ4_NL

# 💡 显存计算公式：
# 总显存 ≈ 模型显存 + KV Cache + 系统开销
# KV Cache ≈ 2GB (8K context)
# 系统开销 ≈ 2GB
```

## 2.模型加载失败

```bash
# 📌 模型加载失败排查

# 检查清单：
# 1. 文件是否存在
ls -lh models/qwen2.5-7b-q4_k_m.gguf

# 2. 文件是否完整（无损坏）
# 重新下载模型

# 3. 路径是否正确
# 使用绝对路径
./llama-cli -m /full/path/to/model.gguf

# 4. 权限是否正确
chmod 644 models/*.gguf

# 5. 磁盘空间是否充足
df -h

# 💡 常见错误：
# "failed to load model" → 文件损坏或路径错误
# "invalid model" → 格式不支持
```

## 3.推理速度慢

```bash
# 📌 推理速度慢排查

# 检查项：
# 1. 是否使用 GPU
./llama-cli -m model.gguf -ngl 99  # 确保 -ngl > 0

# 2. 是否启用 CUDA 编译版本
# 使用预编译版本或从源码编译

# 3. CPU 线程数是否合理
./llama-cli -m model.gguf -t 4  # 核心数的一半

# 4. 是否开启 Flash Attention
./llama-cli -m model.gguf --flash-attn on

# 5. 量化级别是否合适
# Q4_K_M 比 Q6_K 快 15-20%

# 🌟 速度参考值：
# 7B Q4_K_M，RTX 3060 → ~40 tokens/s
# 7B Q4_K_M，RTX 4090 → ~80 tokens/s
# 13B Q4_K_M，RTX 3060 → ~20 tokens/s
```

## 4.输出质量差

```bash
# 📌 输出质量差排查

# 可能原因：
# 1. 量化级别太低
# 解决：使用 Q4_K_M 或更高

# 2. 上下文长度不足
# 解决：增加 -c 值

# 3. 生成参数不合适
# 解决：调整 temperature、top_p 等

# 4. 提示词质量差
# 解决：优化 prompt

# 5. 模型本身能力不足
# 解决：换用更大的模型

# 💡 调试技巧：
# - 对比不同量化版本的输出
# - 记录最佳参数组合
# - 使用 system prompt 引导
```
