## 一、为什么二叉树如此重要？

```
💡 二叉树是几乎所有树形数据结构的基础：

二叉树 → 二叉搜索树 → AVL树 → 红黑树 → B树/B+树（数据库索引）
二叉树 → 哈夫曼树（编码压缩）
二叉树 → 堆（优先队列、堆排序）
二叉树 → 表达式树（编译器）
```

📌 学好二叉树，后面的高级数据结构都是"变体"。

## 二、数据结构定义

```c
// 🌟 带父指针的二叉树节点
typedef struct BiTNode {
    int data;
    struct BiTNode *left;    // 左孩子
    struct BiTNode *right;   // 右孩子
    struct BiTNode *parent;  // 🌟 父指针（方便回溯，不是所有实现都有）
} BiTNode, *BiTree;
```

💡 父指针的作用：在需要从子节点向上查找时（如求最近公共祖先 LCA），没有父指针就需要从根重新遍历，有了父指针可以直接回溯。

## 三、先序创建二叉树

### 1. 核心思路

```
先序输入规则：输入 0 表示空节点

输入序列：1 2 0 0 3 0 0
         ↓ 构建过程：
         1          ← 读到1，创建根节点
        / \
       2   3        ← 递归创建左子树和右子树
      / \ / \
     ∅  ∅ ∅  ∅      ← 读到0，返回NULL
```

### 2. 实现代码

```c
static void CreateBitree(BiTree *Bt)
{
    int x;
    scanf("%d", &x);
    if(x == 0)
    {
        *Bt = NULL;  // 📌 0 表示空节点
    }
    else
    {
        *Bt = (BiTree)malloc(sizeof(BiTNode));
        if(*Bt == NULL) { printf("内存申请失败。\n"); exit(1); }

        (*Bt)->data = x;
        (*Bt)->parent = NULL;  // 初始化父指针

        // 🌟 先序：先创建左子树，再创建右子树
        CreateBitree(&(*Bt)->left);
        if((*Bt)->left != NULL)
            (*Bt)->left->parent = *Bt;  // 设置左孩子的父指针

        CreateBitree(&(*Bt)->right);
        if((*Bt)->right != NULL)
            (*Bt)->right->parent = *Bt;  // 设置右孩子的父指针
    }
}
```

⚠️ 传入 `BiTree *Bt`（二级指针）是因为需要修改指针本身的值（从 NULL 变为新节点地址）。

## 四、三种递归遍历

### 1. 遍历方式对比

```
        1
       / \
      2   3
     / \
    4   5

先序（根→左→右）：1 2 4 5 3
中序（左→根→右）：4 2 5 1 3
后序（左→右→根）：4 5 2 3 1
```

### 2. 代码实现

```c
// 🌟 先序遍历：先访问根，再遍历左子树，最后遍历右子树
static void PreOrder(BiTree Bt)
{
    if(Bt != NULL)
    {
        printf("%d ", Bt->data);   // 访问根节点
        PreOrder(Bt->left);        // 遍历左子树
        PreOrder(Bt->right);       // 遍历右子树
    }
}

// 中序遍历：先遍历左子树，再访问根，最后遍历右子树
static void InOrder(BiTree Bt)
{
    if(Bt != NULL)
    {
        InOrder(Bt->left);
        printf("%d ", Bt->data);
        InOrder(Bt->right);
    }
}

// 后序遍历：先遍历左子树，再遍历右子树，最后访问根
static void PostOrder(BiTree Bt)
{
    if(Bt != NULL)
    {
        PostOrder(Bt->left);
        PostOrder(Bt->right);
        printf("%d ", Bt->data);
    }
}
```

💡 **记忆口诀**：先序、中序、后序的"先中后"指的是**根节点的访问时机**。

## 五、非递归遍历（栈实现）

### 1. 为什么需要非递归？

```
⚠️ 递归遍历的问题：
- 每次递归调用都会在系统栈上压入一帧
- 如果树很深（如 10000 层），会导致栈溢出
- 非递归用自己管理的栈，可以控制栈大小
```

### 2. 非递归先序遍历 — 方法一（经典）

```c
// 🌟 核心：根进栈 → 循环{弹出并访问 → 右进栈 → 左进栈}
static void PreOrder1(BiTNode *b)
{
    BiTNode *p = NULL;
    stacktree st;
    initstack(&st);

    if(b != NULL)
    {
        pushtree(&st, b);                    // 根节点进栈
        while(!stackisempty(&st))            // 栈不空就继续
        {
            poptree(&st, &p);                // 弹出栈顶
            printf("%d ", p->data);          // 访问

            if(p->right != NULL)             // 右孩子先进栈
                pushtree(&st, p->right);     //   ↓ 后进先出
            if(p->left != NULL)              // 左孩子后进栈
                pushtree(&st, p->left);      //   ↓ 先出，优先访问
        }
    }
    destroystack(&st);
}
```

💡 为什么要右先进、左后进？因为栈是**后进先出**（LIFO），左孩子需要先被访问，所以后进栈。

### 3. 非递归先序遍历 — 方法二（Morris 思路）

```c
// 🌟 核心：一路向左走，沿途访问并入栈；到头后弹栈转向右子树
static void preorder2(BiTNode *b)
{
    BiTNode *p = b;
    stacktree st;
    initstack(&st);

    while(p != NULL || !stackisempty(&st))
    {
        while(p != NULL)                     // 一路向左
        {
            printf("%d ", p->data);          // 📌 先序：入栈前访问
            pushtree(&st, p);                // 入栈（记住回来的路）
            p = p->left;
        }
        if(!stackisempty(&st))
        {
            poptree(&st, &p);                // 弹出父节点
            p = p->right;                    // 转向右子树
        }
    }
    destroystack(&st);
}
```

### 4. 非递归中序遍历

```c
// 🌟 与先序方法二几乎一样，只是 printf 的位置不同
static void inorder(BiTNode *root)
{
    BiTNode *p = root;
    stacktree st;
    initstack(&st);

    while(p != NULL || !stackisempty(&st))
    {
        while(p != NULL)
        {
            pushtree(&st, p);                // 入栈
            p = p->left;                     // 一路向左
        }
        if(!stackisempty(&st))
        {
            poptree(&st, &p);
            printf("%d ", p->data);          // 📌 中序：弹出时访问
            p = p->right;
        }
    }
    destroystack(&st);
}
```

📌 **对比**：

| 遍历方式 | printf 位置 | 访问时机 |
|:---|:---|:---|
| 先序（方法二） | `while(p!=NULL)` 内部 | 入栈前 |
| 中序 | `poptree` 之后 | 弹栈时 |

## 六、辅助操作

### 1. 统计节点个数

```c
static int CountNodes(BiTree root)
{
    if(root == NULL) return 0;
    return 1 + CountNodes(root->left) + CountNodes(root->right);
}
// 💡 递归公式：节点数 = 1(自身) + 左子树节点数 + 右子树节点数
```

### 2. 求树深度（后序遍历）

```c
static int PostTreeDepth(BiTree root, int h)
{
    static int maxDepth = 0;  // ⚠️ static 变量，全局最大深度
    if(root == NULL)
    {
        if(h > maxDepth) maxDepth = h;  // 到达空节点时更新最大深度
    }
    else
    {
        PostTreeDepth(root->left, h + 1);   // 递归左子树，深度+1
        PostTreeDepth(root->right, h + 1);  // 递归右子树，深度+1
    }
    return maxDepth;
}
```

💡 为什么用后序遍历求深度？因为需要先知道左右子树的深度，才能计算当前节点的深度。

### 3. 后序遍历销毁树

```c
static BiTree KillTree(BiTree root)
{
    if(root != NULL)
    {
        KillTree(root->left);    // 先销毁左子树
        KillTree(root->right);   // 再销毁右子树
        free(root);              // 🌟 最后释放自己
    }
    return NULL;
}
```

⚠️ **必须后序销毁！** 如果先 `free(root)`，就无法访问 `root->left` 和 `root->right` 了——这就是"先拆脚手架再拆墙"的道理。

## 七、先序遍历结果存入数组

```c
static void preorderinarr(BiTree root, int arr[], int *index)
{
    if(root != NULL)
    {
        arr[*index] = root->data;
        (*index)++;
        preorderinarr(root->left, arr, index);
        preorderinarr(root->right, arr, index);
    }
}

static int getpreorder(BiTree root, int arr[])
{
    int index = 0;
    preorderinarr(root, arr, &index);
    return index;  // 返回节点个数
}
```

💡 将遍历结果存入数组是很多算法的基础操作，比如：序列化、比较两棵树是否相同、从遍历序列重建树等。
